def _rng_snapshot(mode):
    assert mode=='cpu'
    n=np.random.get_state()
    return {'python':random.getstate(),'numpy':[n[0],n[1].tolist(),int(n[2]),int(n[3]),float(n[4])],
            'tpu_rng_supported':False},{'rng_cpu':torch.get_rng_state().clone()}


def _restore(j,t):
    def tuples(x):return tuple(tuples(y) for y in x) if isinstance(x,list) else x
    random.setstate(tuples(j['python']))
    n=j['numpy'];np.random.set_state((n[0],np.array(n[1],dtype=np.uint32),n[2],n[3],n[4]))
    torch.set_rng_state(t['rng_cpu'])


class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.base=torch.nn.Parameter(torch.eye(16),requires_grad=False)
        self.lora_weight=torch.nn.Parameter(torch.randn(16,16)*.01)
    def forward(self,input_ids,attention_mask,use_cache):
        assert use_cache is False
        from types import SimpleNamespace
        return SimpleNamespace(logits=self.base[input_ids]+self.lora_weight[input_ids])


class SyntheticBackend:
    """CPU-only serializer fixture, NOT original safetensors/native checkpoint proof."""
    def __init__(self):
        random.seed(1729);np.random.seed(1729);torch.manual_seed(1729)
        self.model=Model();self.mode='cpu';self.rank=0;self.world_size=1
        self.bindings={'tiny_fixture':True};self.gradient_reduction='identity'
        self.trainable_parameters={'lora_weight':self.model.lora_weight}
        self.optimizer=torch.optim.Adam(self.trainable_parameters.values(),lr=1e-5,
            betas=(.9,.999),eps=1e-8,weight_decay=0,foreach=False)
    def barrier(self):pass
    def global_sum(self,value):return float(value)
    def validate_gradients(self):
        assert all(p.grad is not None and torch.isfinite(p.grad).all() for p in self.trainable_parameters.values())
    def save_checkpoint(self,path,*,identity,cursor,max_total_bytes):
        root=Path(path);root.mkdir(exist_ok=False)
        state={'lora':self.model.lora_weight.detach().clone(),'optimizer':self.optimizer.state_dict(),
               'rng':_rng_snapshot('cpu')}
        with (root/'synthetic.pt').open('xb') as f:torch.save(state,f)
        commit={'identity':identity,'cursor':cursor,'payload_sha256':hashlib.sha256((root/'synthetic.pt').read_bytes()).hexdigest(),
                'SYNTHETIC_CPU_ONLY':True}
        raw=canonical(commit)
        with (root/'COMMITTED.json').open('xb') as f:f.write(raw)
        return {'path':str(root),'commit_sha256':hashlib.sha256(raw).hexdigest(),'world_size':1,
                'prewrite_all_rank_serialized_upper_bytes':1024*1024}
    def load_checkpoint(self,path,*,commit_sha256,identity,cursor):
        root=Path(path);raw=(root/'COMMITTED.json').read_bytes()
        assert hashlib.sha256(raw).hexdigest()==commit_sha256
        m=json.loads(raw);assert m['identity']==identity and m['cursor']==cursor and m['SYNTHETIC_CPU_ONLY']
        assert hashlib.sha256((root/'synthetic.pt').read_bytes()).hexdigest()==m['payload_sha256']
        # Only our newly authored synthetic fixture; no real checkpoint/weights read.
        d=torch.load(root/'synthetic.pt',weights_only=False,map_location='cpu')
        with torch.no_grad():self.model.lora_weight.copy_(d['lora'])
        self.optimizer.load_state_dict(d['optimizer']);self.optimizer.zero_grad(set_to_none=True)
        _restore(*d['rng']);self.model.eval()


def parsed():
    return [{'parent_id':f'synthetic-{i:03d}','train_ordinal':i,
             'supports':[([[i%10]], [[(i+1)%10]])] if i%7==0 else
                        [([[i%10]], [[(i+1)%10]]),([[(i+2)%10,(i+3)%10]], [[(i+4)%10,(i+5)%10]])]}
            for i in range(128)]


def make(root,deadline):
    s=sources();data=fixture_schedule(s,parsed());b=SyntheticBackend()
    return FixtureRuntime(s,data,b,root,'synthetic-public-sft8',deadline)


class RuntimeTests(unittest.TestCase):
    def test_public_exact_schedule_and_all_folds(self):
        s=sources();data=materialize_public_schedule(s,PROJECT,CENSUS)
        self.assertTrue(data.native)
        self.assertEqual(sum(len(e['ids'])-len(e['prompt']) for u in range(1,9) for e in data.examples(u)),122894)
        self.assertEqual(sum(len(e['ids'])-len(e['prompt']) for e in data.monitor()),1572)
        self.assertEqual(len({e['parent_id'] for e in data.examples(1)}),112)
        self.assertEqual(sum(len(e['ids'])-len(e['prompt'])>512 for e in data.examples(3)),1)
        span=span_dispatcher(s,data)
        for u in range(1,9):
            for e in data.examples(u):
                q=span(e['ids'],[1]*len(e['ids']),query_prompt_ids=e['prompt'])
                self.assertEqual(q['target_ids'],e['ids'][len(e['prompt']):])
                self.assertEqual(q['prediction_positions'],list(range(len(e['prompt'])-1,len(e['ids'])-1)))

    def test_zero_demo_bound_and_default_still_rejects(self):
        s=sources();d=fixture_schedule(s,parsed());span=span_dispatcher(s,d)
        ex=next(e for u in range(1,9) for e in d.examples(u) if e['prompt'].count(14)==2)
        with self.assertRaises(ValueError):s.module('query_spans').query_response_view(ex['ids'],[1]*len(ex['ids']),query_prompt_ids=ex['prompt'])
        q=span(ex['ids']+[13]*7,[1]*len(ex['ids'])+[0]*7,query_prompt_ids=ex['prompt'])
        self.assertEqual(q['aligned_labels'],[-100]*len(ex['prompt'])+ex['ids'][len(ex['prompt']):]+[-100]*7)
        bad=copy.deepcopy(ex);bad['ids'][-2]=(bad['ids'][-2]+1)%10
        with self.assertRaises(ValueError):span(bad['ids'],[1]*len(bad['ids']),query_prompt_ids=bad['prompt'])
        with self.assertRaises(ValueError):span(ex['ids']+[13],[1]*len(ex['ids'])+[1],query_prompt_ids=ex['prompt'])

    def test_static_full16_one_shift_matches_manual(self):
        s=sources();d=fixture_schedule(s,parsed());gt=s.module('global_teacher');gt.query_response_view=span_dispatcher(s,d)
        for ex in (d.examples(1)[0],next(e for e in d.examples(1) if e['prompt'].count(14)==2)):
            ids,a,labels,n=gt.static_labels(ex,3072)
            z=torch.randn(1,3072,16,requires_grad=True)
            loss=gt.static_nll_from_logits(z,ids,labels)
            start=len(ex['prompt']);targets=torch.tensor(ex['ids'][start:])
            manual=-z[0,start-1:len(ex['ids'])-1].log_softmax(-1).gather(1,targets[:,None]).mean()
            self.assertTrue(torch.allclose(loss,manual,atol=1e-6,rtol=0))
            grad=torch.autograd.grad(loss,z,retain_graph=True)[0]
            oracle=torch.autograd.grad(manual,z)[0]
            self.assertTrue(torch.equal(grad,oracle))
            self.assertEqual(n,len(targets))

    def test_source_import_ignores_ambient_module(self):
        import types
        previous=sys.modules.get('query_spans');poison=types.ModuleType('query_spans')
        poison.query_response_view=lambda *a,**k:(_ for _ in ()).throw(AssertionError('ambient poison'))
        sys.modules['query_spans']=poison
        try:self.assertIsNot(sources().module('global_teacher').query_response_view,poison.query_response_view)
        finally:
            if previous is None:sys.modules.pop('query_spans',None)
            else:sys.modules['query_spans']=previous

    def test_native_denies_fixture_and_public_data_with_cpu(self):
        s=sources();b=SyntheticBackend()
        with tempfile.TemporaryDirectory() as root:
            with self.assertRaises(ValueError):NativeRuntime(s,fixture_schedule(s,parsed()),b,root,'no',time.time()+90)
            pub=materialize_public_schedule(s,PROJECT,CENSUS)
            with self.assertRaises(ValueError):NativeRuntime(s,pub,b,root,'no',time.time()+90)
            with self.assertRaises(ValueError):NativeRuntime(s,copy.copy(pub),b,root,'no',time.time()+90)

    def test_full_eight_update_mean_of_parent_means_oracle(self):
        with tempfile.TemporaryDirectory() as root:
            r=make(root,time.time()+300);oracle=SyntheticBackend()
            frozen=oracle.model.base.detach().clone()
            for u in range(1,9):
                oracle.optimizer.zero_grad(set_to_none=True)
                for ex in r.schedule.examples(u):
                    ids=torch.tensor([ex['ids']]);z=oracle.model(ids,torch.ones_like(ids),False).logits
                    start=len(ex['prompt']);target=ids[0,start:]
                    value=-z[0,start-1:len(ex['ids'])-1].log_softmax(-1).gather(1,target[:,None]).mean()/112
                    value.backward()
                oracle.optimizer.step()
            r.produce()
            self.assertTrue(torch.equal(r.b.model.lora_weight,oracle.model.lora_weight))
            self.assertTrue(torch.equal(r.b.model.base,frozen))
            self.assertTrue(all(p.grad is None for p in r.b.model.parameters()))
            self.assertEqual(float(r.b.optimizer.state[r.b.model.lora_weight]['step']),8)
            self.assertEqual(sorted(p.name for p in Path(root).glob('checkpoint-*')),['checkpoint-2','checkpoint-4','checkpoint-7'])
            self.assertFalse((Path(root)/'checkpoint-8').exists())
            with self.assertRaises(ValueError):r.produce()

    def test_monitor_empty_and_trained_state_nonmutation(self):

    def test_real_fresh_process_cpu_cold7_to8(self):
        with tempfile.TemporaryDirectory() as root:
            deadline=time.time()+300
            env={**os.environ,'PYTHONPATH':str(PROJECT),'TORCH_DEVICE_BACKEND_AUTOLOAD':'0'}
            for phase in ('produce','consume'):
                run=subprocess.run([sys.executable,str(Path(__file__).resolve()),'--cold',phase,root,str(deadline)],env=env,capture_output=True,text=True,timeout=120)
                self.assertEqual(run.returncode,0,run.stderr)
            result=json.loads((Path(root)/'CONSUMER.rank-0.json').read_bytes())
            self.assertEqual(result['status'],'COMPLETE_SFT8_AND_EXACT_COLD_REPLAY8')
            self.assertFalse(result['native_executed']);self.assertFalse(result['HF_ready'])
            self.assertNotEqual(result['producer_pid'],result['consumer_pid'])
            self.assertEqual(result['final_retained_cursor'],8)
            self.assertEqual(sorted(p.name for p in Path(root).glob('checkpoint-*')),['checkpoint-2','checkpoint-4','checkpoint-7','checkpoint-8'])
            self.assertEqual(len(list(Path(root).glob('MONITOR-*.json'))),5)

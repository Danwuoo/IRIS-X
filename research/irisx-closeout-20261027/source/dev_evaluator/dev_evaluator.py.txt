class AdmissionError(ValueError):
    """Fixed safe error codes only; never print private rows or traceback locals."""


def need(ok, code):
    if not ok:
        raise AdmissionError(code)


def closed(obj, keys, code):
    need(type(obj) is dict and set(obj) == set(keys), code)


def canonical(obj):
    return json.dumps(obj, sort_keys=True, separators=(",", ":"), allow_nan=False, ensure_ascii=True).encode("ascii")


def digest(obj):
    return hashlib.sha256(canonical(obj)).hexdigest()


def sha_ok(value):
    return type(value) is str and len(value) == 64 and all(c in "0123456789abcdef" for c in value)


def strict_json(raw):
    need(type(raw) is bytes and 0 < len(raw) <= 32 * 1024 * 1024, "JSON_BYTE_BOUND")
    def pairs(items):
        result = {}
        for k, v in items:
            need(k not in result, "JSON_DUPLICATE_KEY")
            result[k] = v
        return result
    def bad(_):
        raise AdmissionError("JSON_NONFINITE")
    try:
        value = json.loads(raw, object_pairs_hook=pairs, parse_constant=bad)
        def finite(obj):
            if type(obj) is float:
                need(math.isfinite(obj), "JSON_NONFINITE")
            elif type(obj) is list:
                for item in obj: finite(item)
            elif type(obj) is dict:
                for item in obj.values(): finite(item)
        finite(value)
        return value
    except (UnicodeDecodeError, json.JSONDecodeError, RecursionError):
        raise AdmissionError("JSON_PARSE") from None


def grid_body(grid):
    need(type(grid) is list and 1 <= len(grid) <= 30, "GRID_HEIGHT")
    need(all(type(row) is list and 1 <= len(row) <= 30 for row in grid), "GRID_WIDTH")
    need(len({len(row) for row in grid}) == 1, "GRID_RECTANGULAR")
    need(all(type(x) is int and 0 <= x <= 9 for row in grid for x in row), "GRID_COLORS")
    return tuple(x for i, row in enumerate(grid) for x in (([10] if i else []) + row))


def body_grid(body):
    need(type(body) in (list, tuple) and bool(body), "BODY_NONEMPTY")
    rows = [[]]
    for x in body:
        need(type(x) is int and 0 <= x <= 10, "BODY_TOKENS")
        if x == 10: rows.append([])
        else: rows[-1].append(x)
    need(grid_body(rows) == tuple(body), "BODY_CANONICAL")
    return rows


def response(grid):
    return grid_body(grid) + (15,)


def parse_prompt(tokens):
    need(type(tokens) is list and 0 < len(tokens) <= 2560 and all(type(x) is int and 0 <= x < 16 for x in tokens), "PROMPT_TOKENS_CONTEXT")
    need(tokens[-3:] == [14, 12, 10], "PROMPT_QUERY_HEADER")
    frames = []
    at = 0
    while at < len(tokens) - 3:
        role = 11 if len(frames) % 2 == 0 else 12
        need(tokens[at:at + 3] == [14, role, 10], "PROMPT_ALTERNATING_ROLE")
        end = at + 3
        while end < len(tokens) and tokens[end] != 15:
            end += 1
        need(end < len(tokens), "PROMPT_FRAME_EOS")
        frames.append(body_grid(tokens[at + 3:end]))
        at = end + 1
    need(at == len(tokens) - 3 and len(frames) >= 3 and len(frames) % 2 == 1, "PROMPT_PUBLIC_SUPPORT_QUERY")
    return frames

def validate_identity(data, *, expected_sha256):
    closed(data, {"schema", "source_kind", "profile", "role", "origin", "clean_holdout",
        "assignment_sha256", "original_public_input_sha256", "original_roles_sha256",
        "original_selector_source_sha256", "original_generation_source_sha256", "rows"}, "IDENTITY_SCHEMA")
    need(data["schema"] == "new-ps-design-seen-dev32-public-identity/v1" and data["source_kind"] == SOURCE_KIND, "IDENTITY_VERSION")
    need(data["profile"] in ("SYNTHETIC_ONLY", "NATIVE_UNRELEASED") and data["role"] == "DEV" and
         data["origin"] == ORIGIN and data["clean_holdout"] is False, "ORIGINAL_DESIGN_SEEN_DEV_ONLY")
    need(sha_ok(expected_sha256) and digest(data) == expected_sha256, "IDENTITY_EXTERNAL_HASH")
    for key in ("assignment_sha256", "original_public_input_sha256", "original_roles_sha256",
                "original_selector_source_sha256", "original_generation_source_sha256"):
        need(sha_ok(data[key]), "IDENTITY_SOURCE_PINS")
    need(type(data["rows"]) is list and len(data["rows"]) == 32, "EXACT_ORIGINAL_DEV32")
    seen = set(); rows = []
    for ordinal, row in enumerate(data["rows"]):
        closed(row, {"parent_id", "dev_ordinal", "query_demo_ordinal", "canonical_prompt",
                     "canonical_prompt_sha256", "query_input_sha256"}, "PUBLIC_DEV_ROW_SCHEMA")
        pid = row["parent_id"]
        need(type(pid) is str and bool(pid) and pid not in seen, "PUBLIC_PARENT_ID_UNIQUE")
        need(type(row["dev_ordinal"]) is int and row["dev_ordinal"] == ordinal and
             type(row["query_demo_ordinal"]) is int and row["query_demo_ordinal"] >= 0, "ORIGINAL_DEV_ORDER_FOLD")
        if data["profile"] == "SYNTHETIC_ONLY":
            need(pid.startswith("synthetic-dev-"), "SYNTHETIC_ID_FENCE")
        else:
            need(len(pid) == 8 and all(c in "0123456789abcdef" for c in pid), "NATIVE_ORIGINAL_TASK_ID")
        frames = parse_prompt(row["canonical_prompt"])
        need(row["query_demo_ordinal"] < (len(frames) - 1) // 2 + 1, "QUERY_FOLD_WITHIN_PUBLIC_DEMONSTRATION_COUNT")
        need(row["canonical_prompt_sha256"] == digest(row["canonical_prompt"]) and
             row["query_input_sha256"] == digest(frames[-1]), "PUBLIC_PROMPT_QUERY_HASH")
        seen.add(pid)
        rows.append((pid, ordinal, row["query_demo_ordinal"], row["canonical_prompt_sha256"], row["query_input_sha256"]))
    return PublicIdentity(data["profile"], expected_sha256, data["assignment_sha256"],
                          data["original_public_input_sha256"], data["original_selector_source_sha256"], tuple(rows))

def _draw(value, name, ordinal, slot):
    closed(value, {"draw_id", "view_id", "seed", "call_id", "status", "grid", "finished_eos",
                   "truncated", "sampled_logps", "actual_sampled_tokens", "elapsed_seconds"}, "DRAW_SCHEMA")
    need(value["draw_id"] == str(slot) and value["view_id"] == "identity" and
         value["call_id"] == f"{name}:DEV:{ordinal}:{slot}" and type(value["seed"]) is int and
         value["seed"] == 1729 + ordinal * 4096 + slot, "DRAW_ORIGINAL_SLOT_SEED_VIEW")
    status = value["status"]
    need(status in ("VALID_EOS", "KNOWN_INVALID_CAP", "UNKNOWN"), "DRAW_STATUS")
    if status == "UNKNOWN":
        need(all(value[k] is None for k in ("grid", "finished_eos", "truncated", "sampled_logps",
                                            "actual_sampled_tokens", "elapsed_seconds")), "UNKNOWN_NOT_INVALID")
        return _Draw(status, None, None, None)
    need(type(value["finished_eos"]) is bool and type(value["truncated"]) is bool, "TERMINATION_TYPES")
    logps = value["sampled_logps"]
    need(type(logps) is list and 1 <= len(logps) <= 512 and all(type(v) in (float, int) and math.isfinite(v)
         and v <= 1e-6 for v in logps), "RAW_MASKED_LOGPS_FINITE_PROVENANCE_ONLY")
    need(type(value["actual_sampled_tokens"]) is int and value["actual_sampled_tokens"] == len(logps), "SAMPLED_TOKEN_COUNT")
    elapsed = value["elapsed_seconds"]
    need(type(elapsed) in (float, int) and math.isfinite(elapsed) and elapsed >= 0, "MEASURED_DRAW_COST")
    if status == "KNOWN_INVALID_CAP":
        need(value["finished_eos"] is False and value["truncated"] is True and value["grid"] is None
             and len(logps) == 512, "KNOWN_COMPLETE_INVALID_CAP")
        tokens = None
    else:
        need(value["finished_eos"] is True and value["truncated"] is False, "VALID_EOS_TERMINATION")
        tokens = response(value["grid"])
        need(len(tokens) == len(logps) <= 512, "VALID_GENERATED_BODY_EOS_CAP")
    return _Draw(status, tokens, float(elapsed), len(logps))

def frozen_vote_at_one(draws):
    """Original vote: descending exact-grid alias count, then first draw ordinal.

    With G2 this is earliest valid slot whenever both distinct grids tie. Legal
    masked likelihood never influences vote. Empty valid pool selects nothing.
    """
    groups = {}
    for ordinal, draw in enumerate(draws):
        if draw.status == "VALID_EOS":
            entry = groups.setdefault(draw.tokens, [0, ordinal])
            entry[0] += 1
    if not groups:
        return None
    return min(groups, key=lambda tokens: (-groups[tokens][0], groups[tokens][1]))

             (row["parent_id"], row["dev_ordinal"], row["query_demo_ordinal"], row["query_input_sha256"]) ==
             (ident[0], ident[1], ident[2], ident[4]), "TARGET_EXACT_PARENT_ORDER_QUERY_FOLD_INPUT")
        # Scoring truth is not a generated/training response. No gold512/context
        # filter: a valid30x30 truth can have930 bodyEOS tokens and score allwrong.
        answers.append(response(row["grid"]))
    return PrivateEvaluationCase(predictions, tuple(answers))

def _evaluate_admitted_case(case):
    """Private frozen math; no target/file reader or public native release entry."""
    need(type(case) is PrivateEvaluationCase and case.predictions.complete and len(case._targets) == 32, "COMPLETE_PRIVATE_DEV_CASE_REQUIRED")
    profile = "SYNTHETIC_ONLY" if case.predictions.identity.profile == "SYNTHETIC_ONLY" else "NATIVE_RECONSTRUCTION"
    policies = {}; bits = {}
    for name, draws, generation_seconds in zip(ARMS, case.predictions._draws, case.predictions.generation_seconds):
        correct = []; coverage = 0; empty = 0; invalid = 0
        for row, truth in zip(draws, case._targets):
            selected = frozen_vote_at_one(row)
            correct.append(int(selected == truth))
            coverage += int(any(c.tokens == truth for c in row if c.status == "VALID_EOS"))
            empty += int(selected is None)
            invalid += sum(c.status == "KNOWN_INVALID_CAP" for c in row)
        bits[name] = correct
        policies[name] = {"vote_at_one_correct": sum(correct), "parent_denominator": 32,
            "vote_at_one_accuracy": sum(correct) / 32, "oracle_coverage_descriptive_only": coverage,
            "empty_valid_pool_parents": empty, "known_invalid_draws": invalid, "unknown_draws": 0,
            "draw_denominator": 64, "measured_generation_seconds": generation_seconds}
    paired = {}
    for comparator in ("CE", "fresh_base"):
        wins = sum(a > b for a, b in zip(bits["DPO"], bits[comparator]))
        losses = sum(a < b for a, b in zip(bits["DPO"], bits[comparator]))
        paired["DPO_vs_" + comparator] = {"wins": wins, "losses": losses, "ties": 32 - wins - losses,
            "net": wins - losses, "parent_denominator": 32, "delta_vote_at_one": (wins - losses) / 32}
    positive = all(pair["net"] > 0 for pair in paired.values())
    return {"schema": "new-ps-design-seen-dev32-aggregate/v1", "source_kind": SOURCE_KIND,
        "profile": profile, "origin": ORIGIN, "clean_holdout": False, "quality_claim": False,
        "fixed_metric": "vote@1; distinct-grid vote count descending then earliest draw ordinal",
        "policies": policies, "paired": paired, "parent_denominator": 32, "total_fresh_draws": 192,
        "exploratory_positive": positive,
        "outcome": "EXPLORATORY_POSITIVE_REQUIRES_INDEPENDENT_CONFIRMATION" if positive else "STOP_FIXED_DIAGNOSTIC_NO_TUNING",
        "significance_gate_applied": False, "supply96_gate_changed": False,
        "cost": {"new_stage_whole_wall_seconds": case.predictions.whole_wall_seconds,
                 "historical_acquisition_seconds_sunk_but_reported": case.predictions.historical_acquisition_seconds,
                 "historical_cost_not_claimed_free": True,
                 "science_non_generation_TF": 666, "science_backward_calls": 296,
                 "engineering_probe_forward_calls": case.predictions.extra_probe_counts[0],
                 "checkpoint_probe_forward_calls": case.predictions.extra_probe_counts[1],
                 "total_non_generation_forward_calls": 666 + sum(case.predictions.extra_probe_counts),
                 "engineering_checkpoint_seconds_included_in_wholewall": case.predictions.engineering_checkpoint_seconds,
                 "total_pipeline_seconds_including_sunk_acquisition": case.predictions.whole_wall_seconds + case.predictions.historical_acquisition_seconds},
        "native_correctness_computed": profile == "NATIVE_RECONSTRUCTION", "per_parent_correctness_exported": False}

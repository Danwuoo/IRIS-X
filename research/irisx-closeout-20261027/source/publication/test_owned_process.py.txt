class OwnedValidationTests(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory(prefix="SYNTHETIC_OWNED_VALIDATION_"); self.addCleanup(self.temp.cleanup)
        self.root = Path(self.temp.name)
        self.guard = owned.load_guard(SUPERVISOR)

    def kwargs(self):
        return dict(deadline=time.time() + 30, monotonic_deadline=time.monotonic() + 30,
                    child_env=dict(SYNTHETIC_ENV), guard=self.guard)

    def assert_pre_spawn_denied(self, kwargs, code, *, argv=None):
        with mock.patch.object(self.guard, "_OwnedChildScope", side_effect=AssertionError("SYNTHETIC_FORBIDDEN_SPAWN")):
            with self.assertRaisesRegex(ValueError, code):
                owned.run_owned(argv or fixture_command("raise AssertionError('must not execute')"), self.root / "attempt", **kwargs)
        self.assertFalse((self.root / "attempt").exists())

    def test_epoch_deadline_and_cleanup_reserve_denied_before_exec_or_output(self):
        for delta in (-1, 0, 9.9, 10):
            with self.subTest(delta=delta):
                kw = self.kwargs(); kw["deadline"] = time.time() + delta
                self.assert_pre_spawn_denied(kw, "OWNED_ORIGINAL_DEADLINE_CLEANUP_RESERVE")

    def test_monotonic_deadline_cannot_be_extended_by_epoch_clock(self):
        kw = self.kwargs(); kw["monotonic_deadline"] = time.monotonic() + 9
        self.assert_pre_spawn_denied(kw, "OWNED_NONEXTENDABLE_MONOTONIC_DEADLINE")

    def test_deadlines_must_be_finite_no_unbounded_native_stage(self):
        for field in ("deadline", "monotonic_deadline"):
            for value in ((math.inf, -math.inf, math.nan, True, None) if field == "deadline" else (math.inf, -math.inf, math.nan, True)):
                with self.subTest(field=field, value=value):
                    kw = self.kwargs(); kw[field] = value
                    self.assert_pre_spawn_denied(kw, "OWNED_.*DEADLINE|OWNED_FINITE")

    def test_invalid_argv_env_and_log_caps_denied_before_spawn(self):
        for value in ([], [""], [True], "python"):
            with self.subTest(argv=value):
                with mock.patch.object(self.guard, "_OwnedChildScope", side_effect=AssertionError("spawn")):
                    with self.assertRaisesRegex(ValueError, "OWNED_ARGV"):
                        owned.run_owned(value, self.root / "attempt", **self.kwargs())
        for field, value, code in (("child_env", None, "OWNED_CLOSED_ENV"), ("child_env", {"X": 1}, "OWNED_CLOSED_ENV"),
                                   ("log_cap", 0, "OWNED_LOG_BOUND"), ("log_cap", True, "OWNED_LOG_BOUND"),
                                   ("log_cap", 64 * 1024**2 + 1, "OWNED_LOG_BOUND")):
            kw = self.kwargs(); kw[field] = value
            self.assert_pre_spawn_denied(kw, code)

    def test_output_path_symlink_and_existing_attempt_never_overwritten(self):
        target = self.root / "SYNTHETIC_TARGET"; target.mkdir()
        link = self.root / "SYNTHETIC_LINK"; link.symlink_to(target, target_is_directory=True)
        with self.assertRaisesRegex(ValueError, "OWNED_OUTPUT_PATH"):
            owned.run_owned(fixture_command("pass"), link / "attempt", **self.kwargs())
        attempt = self.root / "attempt"; attempt.mkdir(); marker = attempt / "SYNTHETIC_KEEP"; marker.write_bytes(b"UNCHANGED")
        with mock.patch.object(self.guard, "_OwnedChildScope", side_effect=AssertionError("spawn")):
            with self.assertRaises(FileExistsError): owned.run_owned(fixture_command("pass"), attempt, **self.kwargs())
        self.assertEqual(marker.read_bytes(), b"UNCHANGED")

    def test_public_stdin_type_and_cap_denied_before_child(self):
        for index, payload in enumerate(("not-bytes", b"x" * (4 * 1024**2 + 1))):
            with self.subTest(index=index), mock.patch.object(self.guard, "_OwnedChildScope", side_effect=AssertionError("spawn")):
                with self.assertRaisesRegex(ValueError, "OWNED_PUBLIC_STDIN_BOUND"):
                    owned.run_owned(fixture_command("pass"), self.root / ("attempt_" + str(index)), stdin_payload=payload, **self.kwargs())

    def test_guard_load_exact_public_bytes_and_no_side_effects(self):
        raw = SUPERVISOR.read_bytes(); self.assertEqual(digest(raw), owned.SUPERVISOR_SHA)
        with mock.patch.object(subprocess, "Popen", side_effect=AssertionError("import must not spawn")):
            guard = owned.load_guard(SUPERVISOR)
        self.assertTrue(callable(guard._stop_owned_child)); self.assertTrue(callable(guard._OwnedChildScope))
        self.assertIn("PR_SET_PDEATHSIG_FAILED", guard.PARENT_DEATH_GUARD_SOURCE)
        p = self.root / "SYNTHETIC_TAMPERED_SUPERVISOR.py"; p.write_bytes(raw + b"\nraise AssertionError('MUST_NOT_EXECUTE')\n")
        with self.assertRaisesRegex(ValueError, "OWNED_GUARD_EXACT_SOURCE"): owned.load_guard(p)
        link = self.root / "SYNTHETIC_GUARD_LINK.py"; link.symlink_to(SUPERVISOR)

    def test_json_receipt_exclusive0600_and_nonfinite_refusal(self):
        path = self.root / "SYNTHETIC_RECEIPT.json"
        value = {"synthetic": True, "status": "NO_NATIVE_ACTION"}

        self.assertEqual(json.loads(raw), value); self.assertEqual(stat.S_IMODE(path.stat().st_mode), 0o600)

        with self.assertRaises(ValueError): owned.write_json(self.root / "SYNTHETIC_NAN.json", {"x": math.nan})


class ActualHarmlessChildTests(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory(prefix="SYNTHETIC_OWNED_CHILD_"); self.addCleanup(self.temp.cleanup)
        self.root = Path(self.temp.name); self.guard = owned.load_guard(SUPERVISOR)

    def run_child(self, code, *, name="attempt", stdin_payload=None, seconds=25, log_cap=1024**2, env=None):
        return owned.run_owned(fixture_command(code), self.root / name, deadline=time.time() + seconds,
            monotonic_deadline=time.monotonic() + seconds, child_env=dict(SYNTHETIC_ENV if env is None else env),
            guard=self.guard, stdin_payload=stdin_payload, log_cap=log_cap)

    def test_exact_public_stdin_stdout_closed_env_and_receipts(self):
        payload = b"SYNTHETIC_PUBLIC_ONLY\nsecond-line\x00tail"
        code = "import os,sys;data=sys.stdin.buffer.read();assert 'SYNTHETIC_PARENT_ONLY' not in os.environ;sys.stdout.buffer.write(b'RECEIVED:'+data)"
        with mock.patch.dict(os.environ, {"SYNTHETIC_PARENT_ONLY": "NOT_A_REAL_SECRET"}):
            receipt = self.run_child(code, stdin_payload=payload)
        self.assertEqual(receipt["returncode"], 0); self.assertIsNone(receipt["reason"])
        self.assertEqual((self.root / "attempt/PRIVATE-stdout.log").read_bytes(), b"RECEIVED:" + payload)
        self.assertTrue(receipt["wait_returned"]); self.assertFalse(receipt["resource_released_claimed"])
        self.assertFalse(receipt["provider_runtime_limit_proven"]); self.assertFalse(receipt["retry_allowed"])
        self.assertEqual(json.loads((self.root / "attempt/SUPERVISOR.json").read_bytes()), receipt)
        self.assertEqual(stat.S_IMODE((self.root / "attempt").stat().st_mode), 0o700)
        for name in ("PRIVATE-stdout.log", "SUPERVISOR.json"):
            self.assertEqual(stat.S_IMODE((self.root / "attempt" / name).stat().st_mode), 0o600)

    def test_optional_monotonic_deadline_derives_only_remaining_original_wall(self):
        epoch = time.time() + 25; expected = time.monotonic() + 25
        receipt = owned.run_owned(fixture_command("pass"), self.root / "attempt", deadline=epoch,
                                  child_env=dict(SYNTHETIC_ENV), guard=self.guard)
        self.assertEqual(receipt["returncode"], 0)
        self.assertEqual(receipt["original_deadline_epoch"], epoch)
        self.assertTrue(math.isfinite(receipt["original_monotonic_deadline"]))
        self.assertLess(abs(receipt["original_monotonic_deadline"] - expected), .1)

    def test_nonzero_child_is_preserved_not_retried_or_promoted_success(self):
        receipt = self.run_child("import sys;print('SYNTHETIC_FAILURE');sys.exit(7)")
        self.assertEqual(receipt["returncode"], 7); self.assertFalse(receipt["retry_allowed"])
        self.assertEqual((self.root / "attempt/PRIVATE-stdout.log").read_text(), "SYNTHETIC_FAILURE\n")

    def test_log_cap_reports_actual_overshoot_without_claiming_hard_byte_limit(self):
        receipt = self.run_child("import sys;sys.stdout.write('SYNTHETIC_'+('x'*2048))", log_cap=128)
        self.assertEqual(receipt["reason"], "OWNED_LOG_CAP")
        self.assertGreater(receipt["log_bytes"], 128)
        self.assertEqual(receipt["log_bytes"], (self.root / "attempt/PRIVATE-stdout.log").stat().st_size)

    def test_timeout_honors_original_clock_and_stops_harmless_child(self):
        started = time.monotonic()
        receipt = self.run_child("import time;print('SYNTHETIC_SLEEP',flush=True);time.sleep(30)", seconds=10.5)
        self.assertEqual(receipt["reason"], "OWNED_ORIGINAL_DEADLINE")
        self.assertLess(receipt["returncode"], 0); self.assertTrue(receipt["wait_returned"])
        self.assertLess(time.monotonic() - started, 8)

class OriginalInstallerSourceOnlyTests(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory(prefix="SYNTHETIC_INSTALLER_SOURCE_ONLY_"); self.addCleanup(self.temp.cleanup)
        self.root = Path(self.temp.name); self.before_modules = set(sys.modules); self.before_path = list(sys.path)
        self.addCleanup(self.clean_private_imports)
        self.guard = owned.load_guard(SUPERVISOR)
        self.deadline = time.time() + 30; self.mono = time.monotonic() + 30
        self.source = installers.OriginalInstallerSources(ORIGINAL, guard=self.guard, deadline=self.deadline, monotonic_deadline=self.mono)

    def clean_private_imports(self):
        for name in set(sys.modules) - self.before_modules:
            if name.startswith("_new_ps_offline_"): sys.modules.pop(name, None)
        sys.path[:] = self.before_path  # recovered offline module adds runtime path

    def copied_source(self):
        target = self.root / ("SYNTHETIC_COPY_" + str(len(list(self.root.iterdir())))); target.mkdir()
        for rel, raw in self.source.blobs.items():
            p = target / rel; p.parent.mkdir(parents=True, exist_ok=True); p.write_bytes(raw)
        return target

    def make_source(self, root):
        return installers.OriginalInstallerSources(root, guard=self.guard, deadline=self.deadline, monotonic_deadline=self.mono)

    def test_exact_seven_original_public_source_bytes_no_material_or_targets(self):
        self.assertEqual(set(self.source.blobs), set(installers.PINS)); self.assertEqual(len(self.source.blobs), 7)
        for rel, raw in self.source.blobs.items():
            self.assertLess(len(raw), 65536); self.assertEqual(digest(raw), installers.PINS[rel])
        self.assertEqual(set(self.source.modules), {"local_supervisor"})

    def test_imports_bytecompiled_private_namespace_no_spawn_or_prepare(self):
        with mock.patch.object(subprocess, "Popen", side_effect=AssertionError("SOURCE_IMPORT_CANNOT_EXECUTE")), \
             mock.patch.object(installers, "run_owned", side_effect=AssertionError("NO_INSTALLER_EXECUTION")):
            for name in installers.MODULES:
                module = self.source.module(name)
                self.assertIsNone(module.__cached__); self.assertTrue(module.__name__.startswith("_new_ps_offline_"))
                self.assertEqual(Path(module.__file__), ORIGINAL / installers.MODULES[name])
                self.assertIs(self.source.module(name), module)
        self.assertNotIn("torch", set(sys.modules) - self.before_modules)
        self.assertEqual(self.source.module("offline_bootstrap").supervise.__self__, self.source)
        self.assertEqual(self.source.module("mounted_materials").EXPECTED["torch"], "2.8.0+cu128")

    def test_unregistered_module_and_reused_private_namespace_denied(self):
        with self.assertRaisesRegex(ValueError, "INSTALLER_CLOSED_MODULES"): self.source.module("SYNTHETIC_UNREGISTERED")
        self.source.module("portable_interpreter")
        second = self.make_source(ORIGINAL)
        with self.assertRaisesRegex(ValueError, "INSTALLER_FRESH_PRIVATE_NAMESPACE"): second.module("portable_interpreter")

    def test_missing_tampered_oversized_or_symlink_source_fails_before_compile(self):
        for mode in ("missing", "tamper", "oversize", "symlink"):
            with self.subTest(mode=mode):
                root = self.copied_source(); path = root / "portable_interpreter.py"
                if mode == "missing": path.unlink()
                elif mode == "tamper": path.write_bytes(path.read_bytes() + b"\nraise AssertionError('NO_COMPILE')\n")
                elif mode == "oversize": path.write_bytes(b"x" * 65536)
                else: path.unlink(); path.symlink_to(ORIGINAL / "portable_interpreter.py")
                with self.assertRaisesRegex(ValueError, "INSTALLER_(BOUNDED_SOURCE|EXACT_ORIGINAL_BYTES)"): self.make_source(root)
        link = self.root / "SYNTHETIC_ROOT_LINK"; link.symlink_to(ORIGINAL, target_is_directory=True)
        with self.assertRaisesRegex(ValueError, "INSTALLER_SOURCE_ROOT"): self.make_source(link)

    def test_prepare_child_command_is_only_compiled_not_executed_and_pins_preserved(self):
        original = [sys.executable, "-I", str(ORIGINAL / "prepare_python_child.py"), "--assets", "SYNTHETIC_NO_ASSETS",
                    "--manifest-sha", "f" * 64, "--destination", "SYNTHETIC_NO_DESTINATION"]
        rewritten = self.source.guarded_interpreter_entry(original)
        self.assertEqual(rewritten[:5], [sys.executable, "-I", "-S", "-B", "-c"])
        self.assertEqual(rewritten[6:], original[3:]); code = rewritten[5]
        compile(code, "SYNTHETIC_COMPILE_ONLY_NEVER_EXECUTE_PREPARE", "exec")
        self.assertIn(installers.PINS["portable_interpreter.py"], code)
        self.assertIn(installers.PINS["prepare_python_child.py"], code)
        self.assertIn("exec(compile", code); self.assertIn("SOURCE_SYMLINK", code)

    def test_only_reviewed_venv_pip_flags_preserved_without_running(self):
        for module in ("venv", "pip"):
            command = [sys.executable, "-I", "-m", module, "SYNTHETIC_ARGUMENT", "--no-index"]
            self.assertEqual(self.source.guarded_interpreter_entry(command), [sys.executable, "-B", *command[1:]])
        for command in ([sys.executable, "-c", "print('NO')"], [sys.executable, "-I", "-m", "http.server"], ["/bin/sh", "-c", "false"]):
            with self.assertRaisesRegex(ValueError, "INSTALLER_ONLY_REVIEWED_PYTHON_COMMANDS"):
                self.source.guarded_interpreter_entry(command)

    def test_supervise_adapter_preserves_original_deadlines_no_allowance_restart(self):
        argv = [sys.executable, "-I", "-m", "pip", "SYNTHETIC_NOT_AN_INSTALL"]
        with mock.patch.object(installers, "run_owned", return_value={"SYNTHETIC_INERT": True}) as run:
            result = self.source.supervise(argv, self.root / "never_created", max_seconds=1, child_env={})
        self.assertEqual(result, {"SYNTHETIC_INERT": True}); self.assertFalse((self.root / "never_created").exists())
        self.assertEqual(run.call_args.kwargs["deadline"], self.deadline)
        self.assertEqual(run.call_args.kwargs["monotonic_deadline"], self.mono)
        for seconds in (0, 7201, True, None):
            with mock.patch.object(installers, "run_owned", side_effect=AssertionError("invalid allowance reached runner")):
                with self.assertRaisesRegex(ValueError, "INSTALLER_NO_NEW_TIME_ALLOWANCE"):
                    self.source.supervise(argv, self.root / "never_created", max_seconds=seconds, child_env={})

    def test_prepare_remains_explicit_default_off_without_calling_it(self):
        tree = ast.parse((ACTIVE / "publication/installer_sources.py").read_bytes())
        cls = next(n for n in tree.body if isinstance(n, ast.ClassDef) and n.name == "OriginalInstallerSources")
        method = next(n for n in cls.body if isinstance(n, ast.FunctionDef) and n.name == "prepare")
        self.assertEqual([a.arg for a in method.args.kwonlyargs], ["execute"])
        self.assertIs(method.args.kw_defaults[0].value, False)
        self.assertIsInstance(method.body[0], ast.Expr)
        self.assertIn("INSTALLER_DEFAULT_OFF", ast.unparse(method.body[0]))

def need(ok,code):
    if not ok:raise ValueError(code)


def sha(raw):return hashlib.sha256(raw).hexdigest()


def strict_json(raw):
    def pairs(items):
        result={}
        for key,value in items:
            need(key not in result,'WRAPPER_DUPLICATE_JSON_KEY');result[key]=value
        return result
    def invalid(value):raise ValueError('WRAPPER_NONFINITE_JSON')
    return json.loads(raw,object_pairs_hook=pairs,parse_constant=invalid)


def read(path,cap=4*1024**2):
    path=Path(path)
    need(path.is_absolute() and not any(p.is_symlink()for p in(path,*path.parents)),"WRAPPER_NO_SYMLINK")
    need(path.is_file() and 0<path.stat().st_size<=cap,"WRAPPER_PUBLIC_FILE_BOUND")
    return path.read_bytes()

def verified_source_call(path,pin,function,arguments):
    """PUBLIC JSON authority/paths only; private candidates/targets never inline."""
    code=("import hashlib,json,sys\nfrom pathlib import Path\nx=json.loads(sys.stdin.read());p=Path(x['path'])\n"
       "if any(q.is_symlink()for q in(p,*p.parents)):raise SystemExit('SOURCE_SYMLINK')\n"
       "b=p.read_bytes()\nif hashlib.sha256(b).hexdigest()!=x['pin']:raise SystemExit('SOURCE_HASH')\n"
       "n={'__name__':'new_ps_source_verified_stage','__file__':str(p),x['marker']:x['pin']}\n"
       "exec(compile(b,str(p),'exec',dont_inherit=True),n)\n"
       "r=n[x['function']](*x['args'],**x['kwargs'])\n"
       "print(json.dumps({'stage_returned':True,'status':r.get('status','RETURNED')},sort_keys=True))\n")
    data=dict(path=str(path),pin=pin,function=function,**arguments)
    return code,json.dumps(data,sort_keys=True,separators=(",",":")).encode()

def failure_pack(root,stage,deadline,mono):
    """Best effort small PUBLIC metadata only; no private logs/partial predictions."""
    if time.time()>=deadline or time.monotonic()>=mono:return None
    value={"schema":"new-ps-wrapper-safe-failure/v1","status":"FAILED_OR_UNKNOWN_ATTEMPT_CONSUMED",
           "stage":stage,"retry_allowed":False,"provider_resource_release":"UNKNOWN",
           "model_stop_not_inferred_from_wrapper_exit":True,"quality_GO":False}
    path=root/"FAILURE-SAFE.json"
    if path.exists():
        raw=read(path);return {'bytes':len(raw),'sha256':sha(raw)}
    pin=write_json(path,value)
    raw=read(path)
    if time.time()>=deadline or time.monotonic()>=mono:return pin
    zpath=root/"PUBLIC-FAILURE-METADATA.NEW.zip"
    with zipfile.ZipFile(zpath,'x',compression=zipfile.ZIP_DEFLATED)as archive:archive.writestr(path.name,raw)
    os.chmod(zpath,0o600)
    return pin

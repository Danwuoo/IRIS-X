_BUNDLE_KEY=object()

@dataclass(frozen=True)
class VerifiedPublicBundle:
    """Immutable assembly provenance, not a caller's public=True assertion.

    Only assemble_public_bundle mints this after the full closed public source
    allowlist. This is process discipline, not a malicious Python/OS boundary.
    """
    payload:bytes
    manifest_raw:bytes
    root_grant_sha256:str
    _key:object=field(repr=False,compare=False)
    def __post_init__(self):
        need(self._key is _BUNDLE_KEY and type(self.payload)is bytes and type(self.manifest_raw)is bytes,
             'VERIFIED_PUBLIC_ASSEMBLY_FACTORY_REQUIRED')
    @property
    def manifest(self):return json.loads(self.manifest_raw)

def need(ok,code):
    if not ok:raise ValueError(code)

def sha(raw):return hashlib.sha256(raw).hexdigest()

def canonical(value):return json.dumps(value,sort_keys=True,separators=(',',':'),allow_nan=False).encode()+b'\n'

def read(path):
    need(path.is_absolute() and not any(p.is_symlink()for p in(path,*path.parents)) and path.is_file(),
         'PUBLIC_BUNDLE_REGULAR_SOURCE')
    need(0<path.stat().st_size<=4*1024**2,'PUBLIC_BUNDLE_FILE_BOUND')
    return path.read_bytes()

def pin(raw):return {'bytes':len(raw),'sha256':sha(raw)}

def validate_wrapper_manifest(wrapper):
    need(type(wrapper)is dict and set(wrapper)=={'schema','files'} and
         wrapper['schema']=='new-ps-setup-export-source-manifest/v1' and
         type(wrapper['files'])is dict and set(wrapper['files'])=={p for _,p in WRAPPER_MODULES}|WRAPPER_EXTRA,
         'EXACT_WRAPPER_SOURCE_MEMBERS_BEFORE_READ')
    return wrapper

def assemble_public_bundle(project_root,original_root,config_root,wrapper_grant,*,native_source_root=None):
    """Future local packaging only, after exact public config/root pins exist."""
    project,original,config=map(Path,(project_root,original_root,config_root))
    native_project=Path(native_source_root)if native_source_root is not None else project
    need(set(wrapper_grant['files'])=={'WRAPPER-CONTRACT.json','WRAPPER-SOURCE-MANIFEST.json',
        'WRAPPER-INDEPENDENT-REVIEW.json','NATIVE-ROOT-GRANT.json'},'CLOSED_WRAPPER_ROOT_DOCS')
    documents={name:read(config/name)for name in CONFIG_NAMES}
    for name,expected in wrapper_grant['files'].items():need(sha(documents[name])==expected,'WRAPPER_CONFIG_PIN')
    native_grant=json.loads(documents['NATIVE-ROOT-GRANT.json'])
    need(set(native_grant['files'])==CONFIG_NAMES-{'WRAPPER-CONTRACT.json','WRAPPER-SOURCE-MANIFEST.json',
        'WRAPPER-INDEPENDENT-REVIEW.json','NATIVE-ROOT-GRANT.json'},'CLOSED_NATIVE_ROOT_DOCS')
    for name,expected in native_grant['files'].items():need(sha(documents[name])==expected,'NATIVE_CONFIG_PIN')
    need(sha(documents['SOURCE-MANIFEST.json'])==NATIVE_MANIFEST,'EXACT_REVIEWED_NATIVE_V6')
    native=json.loads(documents['SOURCE-MANIFEST.json']);wrapper=json.loads(documents['WRAPPER-SOURCE-MANIFEST.json'])
    validate_wrapper_manifest(wrapper)
    blobs={'config/'+name:raw for name,raw in documents.items()}
    for manifest in(native,wrapper):
        for rel,expected in manifest['files'].items():
            need(PurePosixPath(rel).parts[0]in('native_experiment','training','publication') and
                 '..'not in PurePosixPath(rel).parts and not PurePosixPath(rel).is_absolute(),'SOURCE_ONLY_PREFIX')
            source_root=native_project if manifest is native else project
            raw=read(source_root/rel);need(pin(raw)==expected,'EXACT_PUBLIC_SOURCE_PIN');blobs['source/'+rel]=raw
    # Parse literals ONLY from independently pinned public backend/installer source.
    # No imports/execution or original source directory enumeration.
    def literals(raw,names):
        out={}
        for node in ast.parse(raw).body:
            if isinstance(node,ast.Assign) and len(node.targets)==1 and isinstance(node.targets[0],ast.Name):
                name=node.targets[0].id
                if name in names:
                    value=node.value.args[0]if isinstance(node.value,ast.Call) and isinstance(node.value.func,ast.Name) and node.value.func.id=='MappingProxyType'else node.value
                    out[name]=ast.literal_eval(value)
        need(set(out)==set(names),'PINNED_PUBLIC_LITERAL_TABLES');return out
    installer=literals(blobs['source/publication/installer_sources.py'],{'PINS'})['PINS']
    backend=literals(blobs['source/native_experiment/backend_load.py'],{'SOURCE_PINS','ASSET_PINS'})
    originals=dict(installer)
    originals.update({'runtime/'+rel:expected for rel,expected in backend['SOURCE_PINS'].values()})
    originals.update({'runtime/'+rel:expected for rel,expected in backend['ASSET_PINS'].items()})
    for rel,expected in originals.items():
        raw=read(original/rel);need(sha(raw)==expected,'ORIGINAL_PUBLIC_SOURCE_PIN');blobs['original/'+rel]=raw
    manifest={'schema':'new-ps-public-only-notebook-bundle/v1','files':{k:pin(v)for k,v in sorted(blobs.items())},
      'private_targets_or_pairpack_included':False,'runtime_or_weights_included':False,'old_study_entry_included':False}
    raw_manifest=canonical(manifest);blobs['PUBLIC-BUNDLE-MANIFEST.json']=raw_manifest
    stream=io.BytesIO()
    with zipfile.ZipFile(stream,'w',compression=zipfile.ZIP_DEFLATED)as archive:
        for name,raw in sorted(blobs.items()):
            info=zipfile.ZipInfo(name,date_time=(2026,10,27,0,0,0));info.compress_type=zipfile.ZIP_DEFLATED
            info.external_attr=(0o100600<<16);archive.writestr(info,raw)
    payload=stream.getvalue();need(len(payload)<20_000_000 and sum(map(len,blobs.values()))<=16*1024**2,'PUBLIC_BUNDLE_SMALL_CAP')
    bundle=VerifiedPublicBundle(payload,raw_manifest,sha(canonical(wrapper_grant)),_BUNDLE_KEY)
    # Registration occurs ONLY here, never in the dataclass constructor. Normal
    # dataclasses.replace/copy cannot turn a prior capability into a new payload.
    _ASSEMBLED_BUNDLES[id(bundle)]=(bundle,sha(payload),sha(raw_manifest),bundle.root_grant_sha256)
    return bundle

         bundle.root_grant_sha256==sha(canonical(root_grant)),'EXACT_VERIFIED_PUBLIC_ASSEMBLY_CAPABILITY')

    need(registered is not None and registered[0]is bundle and
         registered[1:]==(sha(bundle.payload),sha(bundle.manifest_raw),bundle.root_grant_sha256),

class RecordingHooks:
    def __init__(self): self.events=[]; self.fail_on=None
    def before(self,event,fields):
        self.events.append(('before',event,dict(fields)))
        if self.fail_on==event: raise RuntimeError('synthetic observer failure')
    def after(self,event,fields): self.events.append(('after',event,dict(fields)))
    def failed(self,event,fields): self.events.append(('failed',event,dict(fields)))

class SyntheticBackend:
    """Tiny handwritten CPU algebra; cannot enter NativeEngine."""
    def __init__(self):
        random.seed(1729); np.random.seed(1729)
        self.__dict__.update(vars(tiny_backend())); self.initial_rng=_rng_snapshot(False)
        self.optimizer=torch.optim.Adam(self.trainable_parameters.values(),**ADAM)
        self.barriers=0; self.checkpoints={}; self.corrupt_restore=None
    def reset_root(self):
        self.model.zero_grad(set_to_none=True)
        with torch.no_grad():
            for n,p in self.trainable_parameters.items(): p.copy_(self.initial_adapter[n])
        self.optimizer.state.clear(); _rng_restore(self.initial_rng,False)
        return {'fixture_only':True,'not_native_reset_evidence':True}
    def barrier(self): self.barriers+=1
    def validate_gradients(self):
        norm=0.
        for p in self.trainable_parameters.values():
            if p.grad is None or not torch.isfinite(p.grad).all(): raise ValueError('missing/nonfinite fixture gradient')
            norm+=float(p.grad.square().sum())
        return {'squared_l2':norm,'synthetic_only':True}
    def save_checkpoint(self,path,*,identity,cursor,max_total_bytes):
        root=Path(path);root.mkdir(); assert max_total_bytes==780*1024**2
        raw=json.dumps({'identity':identity,'cursor':cursor},sort_keys=True).encode()
        (root/'SYNTHETIC-MARKER.json').write_bytes(raw);pin=hashlib.sha256(raw).hexdigest()
        self.checkpoints[path]=(copy.deepcopy(self.model.state_dict()),copy.deepcopy(self.optimizer.state_dict()),_rng_snapshot(False),copy.deepcopy(identity),copy.deepcopy(cursor),pin)
        return {'path':path,'commit_sha256':pin,'synthetic_only':True}
    def load_checkpoint(self,path,*,commit_sha256,identity,cursor):
        state,opt,rng,expected_identity,expected_cursor,pin=self.checkpoints[path]
        assert (identity,cursor,commit_sha256)==(expected_identity,expected_cursor,pin)
        with torch.no_grad():
            for n,p in self.trainable_parameters.items():p.copy_(state[n])
        self.optimizer.load_state_dict(copy.deepcopy(opt));_rng_restore(rng,False)
        if self.corrupt_restore=='adapter':
            with torch.no_grad():next(iter(self.trainable_parameters.values())).flatten()[0].add_(1)
        elif self.corrupt_restore=='rng':random.random()
        elif self.corrupt_restore=='adam':next(iter(self.optimizer.state.values()))['exp_avg'].flatten()[0].add_(1)
        return {'restored':True,'synthetic_only':True}

class SyntheticEngine(_EngineCore):
    def __init__(self,backend=None,hooks=None):
        b=backend or SyntheticBackend()
        if b.mode!='cpu' or b.bindings.get('tiny_fixture')is not True:raise ValueError('synthetic CPU only')
        super().__init__(b,hooks=hooks or RecordingHooks(),deadline=time.time()+300,native=False,root_identity_document={'fixture':'HANDWRITTEN_CPU'})

    def test_actual_nonreentrant_checkpoint_full_replay_is_physically_counted(self):
        from torch.utils.checkpoint import checkpoint
        from native_experiment.physical_audit import AuditHooks
        class Block(torch.nn.Module):
            def __init__(self):
                super().__init__()
                self.head=torch.nn.Parameter(torch.randn(8,16)*.2,requires_grad=False)
                self.lora_A=torch.nn.Parameter(torch.randn(8,32)*.03)
                self.lora_B=torch.nn.Parameter(torch.zeros(32,16))
            def forward(self,x):return x@self.head+(x@self.lora_A)@self.lora_B
        class Model(torch.nn.Module):
            def __init__(self):
                super().__init__();self.embedding=torch.nn.Parameter(torch.randn(16,8)*.2,requires_grad=False);self.block=Block()
            def forward(self,input_ids,attention_mask,use_cache=False):
                x=self.embedding[input_ids]*attention_mask[:,:,None]
                z=checkpoint(self.block,x,use_reentrant=False)if self.training else self.block(x)
                return types.SimpleNamespace(logits=z)
        b=SyntheticBackend();b.model=Model();b.trainable_parameters={n:p for n,p in b.model.named_parameters()if p.requires_grad}
        b.initial_adapter={n:p.detach().clone()for n,p in b.trainable_parameters.items()}
        b.optimizer=torch.optim.Adam(b.trainable_parameters.values(),**ADAM)
        default_backend=copy.deepcopy(b)
        default=CpuFixtureEngine(default_backend);default.reset('ce')
        hooks=AuditHooks(science_deadline=time.time()+60,emit=lambda kind,fields:None,max_root_calls=10,max_root_input_positions=10000,max_layer_calls=20)
        hooks.install(b.model,[b.model.block]);hooks.set_phase('SYNTHETIC_NONREENTRANT_REPLAY')
        e=SyntheticEngine(b,hooks);e.reset('ce');v=e.forward(self.response,grad=True);e.backward(matched_ce_parent(v.selected,1))
        other=default.forward(self.response,grad=True);default.backward(matched_ce_parent(other.selected,1))
        for name,p in e.trainable.items():self.assertTrue(torch.equal(p.grad,default.trainable[name].grad))
        e.commit(1);default.commit(1)
        for name,p in e.trainable.items():self.assertTrue(torch.equal(p,default.trainable[name]))
        snap=hooks.snapshot();self.assertEqual(snap['root_forward_completions'],1)
        self.assertEqual(snap['decoder_layer_execution_completions'],2)
        self.assertEqual(snap['decoder_layer_unknown'],0);self.assertEqual(snap['operation_unknown'],0)
        hooks.close()

"""NEW public-only native inputs/probe preparation. Never reads a private pack."""
from pathlib import Path
import hashlib
import json
from training.contracts import Response, digest
from .contracts import validate_dev_rows, PRIVATE_PIN

ROOT = Path(__file__).resolve().parents[1]


def read(path):
    if any(p.is_symlink() for p in (path, *path.parents)): raise ValueError("PUBLIC_INPUT_SYMLINK")
    return path.read_bytes()


def derive(panel, identity, dev):
    if digest(panel) != "3e95efc3b5fbc9ceb5e14b334da94d785572ec93982b6f12fb332db8f0674a88": raise ValueError("ORIGINAL_PUBLIC_TRAIN_PANEL")
    if digest(identity) != "3fe8315cc8249b4f0e4bb2c15a7da6ebe7799c277fac375c5e1ce032f90fd901": raise ValueError("ORIGINAL_PUBLIC_TRAIN_IDENTITY")
    if digest(dev) != "a6c563a8467c34301ba81785e465f7a9b29679b16398e95c681651a73bf8dcec": raise ValueError("ORIGINAL_PUBLIC_DEV_IDENTITY")
    dev_rows = [{"parent_id": p["parent_id"], "ordinal": p["dev_ordinal"], "query_demo_ordinal": p["query_demo_ordinal"],
                 "prompt": p["canonical_prompt"]} for p in dev["rows"]]
    validate_dev_rows(dev_rows)
    selected = None
    for parent in panel["parents"]:
        tokens = parent["canonical_prompt_tokens"]
        eos = [i for i, value in enumerate(tokens) if value == 15]
        if len(eos) < 5: continue  # two complete support pairs, plus public query input
        start = eos[2] + 4  # second public support assistant body, after14/12/10
        end = eos[3] + 1
        try:
            response = Response(tuple(tokens[:start]), tuple(tokens[start:end]))
        except ValueError:
            continue  # public engineering capacity only, never TRAIN pair selection
        selected = {"prompt": list(response.prompt), "body_EOS": list(response.body_eos), "response_pin": response.pin,
                    "source_parent_id": parent["parent_id"], "source_train_ordinal": parent["train_ordinal"],
                    "source_prompt_sha256": parent["canonical_prompt_sha256"], "source_public_support_position": 1,
                    "selection_rule": "first originalTRAINordinal whose second already-public support pair is a complete Response within engineering caps",
                    "query_gold_used": False, "private_pairpack_used": False}
        break
    if selected is None: raise ValueError("NO_PREREGISTRABLE_PUBLIC_SUPPORT_PROBE")
    return {"schema": "new-ps-native-public-inputs/v1", "source_kind": "NEW_RECONSTRUCTION_NOT_HISTORICAL_FINAL",
            "original_dev_identity_semantic_sha256": digest(dev), "dev_rows": dev_rows,
            "dev_rows_semantic_sha256": digest(dev_rows), "train_identity": identity["rows"],
            "engineering_public_probe": selected, "private_source_bindings": panel["bindings"],
            "expected_opaque_private_pairpack": PRIVATE_PIN, "target_reads": 0, "private_pairpack_reads": 0,
            "design_seen_DEV32": True, "clean_holdout": False}


def main():
    panel = json.loads(read(ROOT / "public_adapter/generated/NORMALIZED-PUBLIC-PANEL.json"))
    identity = json.loads(read(ROOT / "public_adapter/generated/ORIGINAL-TRAIN-IDENTITY.json"))
    dev = json.loads(read(ROOT / "dev_evaluator/ORIGINAL-DESIGN-SEEN-DEV32-PUBLIC.json"))
    result = derive(panel, identity, dev)
    raw = json.dumps(result, sort_keys=True, separators=(",", ":"), allow_nan=False).encode() + b"\n"
    out = ROOT / "native_experiment/NATIVE-PUBLIC-INPUTS.NEW.json"
    with out.open("xb") as stream: stream.write(raw)
    print(json.dumps({"public_inputs_bytes": len(raw), "public_inputs_sha256": hashlib.sha256(raw).hexdigest(),
                     "dev_panel_semantic_sha256": result["dev_rows_semantic_sha256"],
                     "engineering_public_probe_sha256": result["engineering_public_probe"]["response_pin"],
                     "engineering_public_probe_input_positions": len(result["engineering_public_probe"]["prompt"]) + len(result["engineering_public_probe"]["body_EOS"]),
                     "target_reads": 0, "private_pairpack_reads": 0}, indent=2))


if __name__ == "__main__": main()

"""Synthetic byte fixtures only; never opens real native outputs/private inputs."""
from pathlib import Path
import os
import tempfile
import time
import unittest
from unittest import mock
import zipfile
from . import export_results as export


class ExportTests(unittest.TestCase):
    def fixture(self, root):
        p=root/"SYNTHETIC-INPUT.json";p.write_bytes(b'{"synthetic":true}\n');p.chmod(0o600)
        return [(p.name,p,export.pin(p.read_bytes()))]

    def test_small_exact_members_private_mode_and_exclusive_output(self):
        with tempfile.TemporaryDirectory() as tmp:
            root=Path(tmp);root.chmod(0o700);members=self.fixture(root);out=root/"new.zip"
            result=export._zip_verified(out,members,deadline=time.time()+10)
            self.assertEqual(out.stat().st_mode&0o777,0o600)
            self.assertLess(result["bytes"],20_000_000)
            self.assertEqual(export.pin(out.read_bytes()),{k:result[k] for k in("bytes","sha256")})
            with zipfile.ZipFile(out) as archive:
                self.assertEqual(archive.namelist(),[members[0][0]])
                self.assertEqual(archive.read(members[0][0]),members[0][1].read_bytes())
            with self.assertRaises(FileExistsError):export._zip_verified(out,members,deadline=time.time()+10)

    def test_wrong_expected_hash_fails_no_successful_pin(self):
        with tempfile.TemporaryDirectory() as tmp:
            root=Path(tmp);root.chmod(0o700);members=self.fixture(root);members[0][2]["sha256"]="0"*64
            with self.assertRaisesRegex(ValueError,"EXACT_MEMBER_HASH"):
                export._zip_verified(root/"bad.zip",members,deadline=time.time()+10)

    def test_expired_deadline_fails_before_payload_write(self):
        with tempfile.TemporaryDirectory() as tmp:
            root=Path(tmp);root.chmod(0o700)
            with self.assertRaisesRegex(ValueError,"DEADLINE"):
                export._zip_verified(root/"bad.zip",self.fixture(root),deadline=1,clock=lambda:2)

    def test_archive_size_cap_cannot_be_bypassed_by_compression(self):
        with tempfile.TemporaryDirectory() as tmp:
            root=Path(tmp);root.chmod(0o700);members=self.fixture(root)
            with mock.patch.object(export,"CAP",1),self.assertRaisesRegex(ValueError,"20MB|SIZE"):
                export._zip_verified(root/"bad.zip",members,deadline=time.time()+10)

    def test_parent_symlink_and_zip_traversal_denied(self):
        with tempfile.TemporaryDirectory() as tmp:
            root=Path(tmp);root.chmod(0o700);members=self.fixture(root);alias=root/"alias";alias.symlink_to(root,target_is_directory=True)
            with self.assertRaisesRegex(ValueError,"OUTPUT_PATH"):
                export._zip_verified(alias/"bad.zip",members,deadline=time.time()+10)
            with self.assertRaisesRegex(ValueError,"BASENAME"):
                export._zip_verified(root/"bad.zip",[("../bad",members[0][1],members[0][2])],deadline=time.time()+10)

    def test_production_entry_denied_before_any_files_or_private_reads(self):
        with mock.patch.object(export,"read",side_effect=AssertionError("must not read")):
            with self.assertRaisesRegex(ValueError,"ROOT_VERIFIED"):
                export.export_root_authorized("/SYNTHETIC_DO_NOT_READ",{})

    def test_private_export_allowlist_excludes_targets_models_and_checkpoints(self):
        self.assertEqual(set(export.PRIVATE_NAMES),{"PRIVATE-PREDICTIONS-192.json",
            "PRIVATE-OPERATIONS-AND-PREDICTIONS.jsonl","PRIVATE-RESET-RECEIPTS.json"})


if __name__=="__main__":unittest.main()

class V6ExportClockTests(unittest.TestCase):
    def test_export_has_only_original_firstcode_remaining_budget(self):
        c={'first_epoch':100000.,'first_monotonic':200000.}
        seal={'whole_deadline_epoch':107200.,'whole_deadline_monotonic':207200.}
        with mock.patch.object(export.time,'time',return_value=107100.),mock.patch.object(export.time,'monotonic',return_value=207100.):
            epoch,mono,tick=export._export_deadlines(c,seal)
            self.assertEqual((epoch,mono),(107200.,207200.));self.assertEqual(tick(),107100.)
    def test_exact7200_denied_even_with_wallclock_backwards(self):
        c={'first_epoch':100000.,'first_monotonic':200000.}
        seal={'whole_deadline_epoch':107200.,'whole_deadline_monotonic':207200.}
        with mock.patch.object(export.time,'time',return_value=100001.),mock.patch.object(export.time,'monotonic',return_value=207200.):
            with self.assertRaises(ValueError):export._export_deadlines(c,seal)
    def test_attempt_to_reset_preload_deadline_denied(self):
        with mock.patch.object(export.time,'time',return_value=100100.),mock.patch.object(export.time,'monotonic',return_value=200100.):
            with self.assertRaises(ValueError):export._export_deadlines({'first_epoch':100000.,'first_monotonic':200000.},
                {'whole_deadline_epoch':107300.,'whole_deadline_monotonic':207300.})

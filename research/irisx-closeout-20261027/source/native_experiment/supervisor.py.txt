"""NEW bounded native-child supervisor. No direct/default execution or gold.

Parent authenticates this source in the same final reviewed source manifest.
Only a root-released call can launch the model child. Child stdout/stderr stay
private. A model-stop receipt is written only AFTER wait() establishes process
exit, never by the still-running model process itself.

Temporary TERM/HUP/INT handlers request shutdown without raising asynchronous
exceptions across Popen ownership assignment or cleanup. Communication checks
those requests at most every second; every post-spawn exit path has finally
cleanup. Linux child PR_SET_PDEATHSIG(SIGKILL), armed before stdin/bootstrap,
checks expected parent PID before/after arming and covers parent SIGKILL races.
This is not a host/provider liveness guarantee: kernel uninterruptible tasks,
host death, provider billing/resource release and descendants escaping the owned
process group remain UNKNOWN and bounded only by external provider admission.
No stopped receipt is fabricated if bounded wait cannot establish child exit.
"""
from pathlib import Path
import hashlib
import json
import math
import os
import signal
import subprocess

def need(ok, code):
    if not ok: raise ValueError(code)


def sha(raw): return hashlib.sha256(raw).hexdigest()


def _read_public(path, cap=2 * 1024**2):
    need(path.is_absolute() and not any(p.is_symlink() for p in (path, *path.parents)), "SUPERVISOR_PATH")
    need(path.is_file() and 0 < path.stat().st_size <= cap, "SUPERVISOR_FILE_BOUND")
    return path.read_bytes()


def _write(path, value):
    raw = json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False).encode() + b"\n"
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    with os.fdopen(fd, "wb") as stream: stream.write(raw); stream.flush(); os.fsync(stream.fileno())
    fd = os.open(path.parent, os.O_RDONLY | os.O_DIRECTORY)
    try: os.fsync(fd)
    finally: os.close(fd)
    return {"bytes": len(raw), "sha256": sha(raw)}


TERM_GRACE_SECONDS = 5
KILL_GRACE_SECONDS = 5
COMMUNICATE_SLICE_SECONDS = 1.0
TERMINATION_SIGNALS = (signal.SIGTERM, signal.SIGHUP, signal.SIGINT)

# Included literally in the reviewed supervisor source/hash. The native child
# executes this before reading stdin or compiling any bootstrap/model code.
PARENT_DEATH_GUARD_SOURCE = r"""
import ctypes,os,signal,sys

def _arm_parent_death_kill(expected_parent_pid):
    if not sys.platform.startswith('linux') or type(expected_parent_pid) is not int or expected_parent_pid <= 1:
        raise SystemExit('LINUX_VALID_EXPECTED_PARENT_REQUIRED')
    if os.getppid() != expected_parent_pid:
        raise SystemExit('PARENT_CHANGED_BEFORE_PDEATHSIG')
    libc = ctypes.CDLL(None, use_errno=True)
    prctl = libc.prctl
    prctl.argtypes = [ctypes.c_int, ctypes.c_ulong, ctypes.c_ulong, ctypes.c_ulong, ctypes.c_ulong]
    prctl.restype = ctypes.c_int
    if prctl(1, signal.SIGKILL, 0, 0, 0) != 0:
        raise OSError(ctypes.get_errno(), 'PR_SET_PDEATHSIG_FAILED')
    if os.getppid() != expected_parent_pid:
        os.kill(os.getpid(), signal.SIGKILL)
        raise SystemExit('PARENT_CHANGED_DURING_PDEATHSIG')
"""


def _stop_owned_child(process):
    """Only our still-owned process group; finite waits, no false stop claim."""
    if process.poll() is None:
        try: os.killpg(process.pid, signal.SIGTERM)
        except ProcessLookupError: pass
        try: process.wait(timeout=TERM_GRACE_SECONDS)
        except subprocess.TimeoutExpired:
            try: os.killpg(process.pid, signal.SIGKILL)
            except ProcessLookupError: pass
            try: process.wait(timeout=KILL_GRACE_SECONDS)
            except subprocess.TimeoutExpired as error:
                raise RuntimeError('OWNED_CHILD_EXIT_UNOBSERVED_PROVIDER_STATE_UNKNOWN') from error
    else:
        process.wait(timeout=KILL_GRACE_SECONDS)
    need(process.poll() is not None, "OWNED_CHILD_EXIT_REQUIRED")


class _OwnedChildScope:
    """Main-thread signal ownership and unconditional post-spawn cleanup.

    Signal handlers only set a flag: they cannot strand a just-created child
    between Popen returning and assignment. Repeated signals during cleanup are
    ignored, then all prior handlers are restored even when cleanup fails.
    """
    def __init__(self):
        self.process = None
        self.termination_signal = None
        self.cleaning = False
        self.previous_handlers = {}

    def _handle(self, signum, frame):
        if self.termination_signal is None:
            self.termination_signal = signum

    def __enter__(self):
        try:
            for signum in TERMINATION_SIGNALS:
                self.previous_handlers[signum] = signal.getsignal(signum)
                signal.signal(signum, self._handle)
        except BaseException:
            self._restore_handlers()
            raise
        return self

    def _restore_handlers(self):
        for signum, handler in self.previous_handlers.items():
            signal.signal(signum, handler)
        self.previous_handlers.clear()

    def spawn(self, *args, **kwargs):
        need(self.process is None and self.termination_signal is None, 'NO_SPAWN_AFTER_TERMINATION_REQUEST')
        self.process = subprocess.Popen(*args, **kwargs)
        return self.process

    def __exit__(self, exc_type, exc, traceback):
        self.cleaning = True
        try:
            if self.process is not None:
                _stop_owned_child(self.process)
        finally:
            self._restore_handlers()
        return False


def _communicate_until_exit(process, data, *, deadline, scope, clock=None):
    """Polling is only IPC supervision, never model retry or new generation."""
    clock = time.time if clock is None else clock
    while True:
        if scope.termination_signal is not None:
            return 'TERMINATION_REQUESTED'
        remaining = deadline - clock()
        if remaining <= 0:
            return 'SCIENCE_DEADLINE'
        try:
            process.communicate(data, timeout=min(COMMUNICATE_SLICE_SECONDS, remaining))
            return 'EXITED'
        except subprocess.TimeoutExpired:
            data = None  # communicate resumes its buffered input, never resends

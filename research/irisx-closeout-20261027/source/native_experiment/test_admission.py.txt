No private-pack reads or rehashes occur. Existing supply96 failure is immutable.

def raw_json(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False).encode() + b"\n"


def pin(raw):
    return hashlib.sha256(raw).hexdigest()


def paid_account():
    return {"fixture": FIXTURE_RUN, "platform": "PAID_EXISTING_ACCOUNT_SYNTHETIC_ONLY",
            "observed_epoch": NOW - 60, "valid_until_epoch": NOW + 60,

                "run_id": FIXTURE_RUN, "retry_allowed": False},

        frozen_dev = {key: "c" * 64 for key in admission.DEV_PIN_KEYS}

                                                   DEV_gold_attached_to_model=False, design_seen=True)
        self.marker = {"mode": "COMPILE_PINNED_BYTES_NO_PYCACHE", "bootstrap_sha256": BOOTSTRAP_PIN}
        self.inert_result = object()
        self.reseal_synthetic_docs()

    def reseal_synthetic_docs(self):
        """Hashes SYNTHETIC PUBLIC DOCUMENTS ONLY; never the private pairpack."""
        source_pin = pin(raw_json(self.docs["SOURCE-MANIFEST.json"]))
        self.docs["EXPERIMENT-CONTRACT.json"]["source_manifest_sha256"] = source_pin
        contract_pin = pin(raw_json(self.docs["EXPERIMENT-CONTRACT.json"]))
        self.docs["INDEPENDENT-REVIEW.json"].update(source_manifest_sha256=source_pin, contract_sha256=contract_pin)
        self.docs["ROOT-RELEASE.json"]["contract_sha256"] = contract_pin
        self.grant = {"schema": "new-ps-explicit-native-root-grant/v1", "run_id": FIXTURE_RUN,
                      "scope": "ONE_CHARGED_ENGINEERING_THEN_FIXED37_DPO_CE_DEV32_EXPERIMENT",
                      "files": {}, "bootstrap_sha256": BOOTSTRAP_PIN}
        for name in admission.NAMES:
            raw = raw_json(self.docs[name]); (self.root / name).write_bytes(raw)
            self.grant["files"][name] = pin(raw)

    def invoke_inert(self):
        # Both patches are mandatory: these fixture claims can never grant work.
        with ExitStack() as stack:
            stack.enter_context(mock.patch.object(admission.time, "time", return_value=NOW))
            self.source_check = stack.enter_context(mock.patch.object(source_closure, "verify_loaded_source_bytes", return_value=True))
            self.factory = stack.enter_context(mock.patch.object(admission, "NativeRelease", return_value=self.inert_result))
            private_read = stack.enter_context(mock.patch.object(native_entry, "_read_pin", side_effect=AssertionError("PRIVATE/MODEL INPUT FORBIDDEN")))
            model_load = stack.enter_context(mock.patch.object(native_entry, "load_native_backend", side_effect=AssertionError("MODEL LOAD FORBIDDEN")))
            try:
                return admission.issue_native_release(self.root, out_of_band_root_grant=self.grant,
                                                      reviewed_byte_bootstrap=self.marker)
            finally:
                private_read.assert_not_called(); model_load.assert_not_called()


class InertAdmissionTests(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory(prefix="SYNTHETIC_ADMISSION_FIXTURE_")
        self.addCleanup(self.temp.cleanup)
        self.root = Path(self.temp.name)

    def fixture(self, account=None):
        # Each subcase has an independent directory; never retry a consumed lease.
        root = self.root / ("fixture_" + str(len(list(self.root.iterdir()))))
        root.mkdir(mode=0o700)
        return SyntheticAdmissionFixture(root, account)

    def assert_denied(self, fixture, code):
        with self.assertRaisesRegex((ValueError, RuntimeError), code): fixture.invoke_inert()
        fixture.factory.assert_not_called()
        self.assertFalse((fixture.root / "attempt").exists())

    def test_paid_exact150_validates_only_inert_factory_and_consumes_one_lease(self):
        f = self.fixture(); self.assertIs(f.invoke_inert(), f.inert_result)
        f.factory.assert_called_once(); f.source_check.assert_called_once()
        attempt = f.root / "attempt"; seal = attempt / "PRELOAD-SEAL.json"
        self.assertEqual(stat.S_IMODE(attempt.stat().st_mode), 0o700)
        self.assertEqual(stat.S_IMODE(seal.stat().st_mode), 0o600)
        value = json.loads(seal.read_bytes())
        self.assertTrue(value["attempt_consumed"]); self.assertFalse(value["retry_allowed"])
        self.assertFalse(value["model_loaded"]); self.assertFalse(value["private_pack_opened"])
        self.assertTrue(value["engineering_gate_still_required"])
        pins, release_pin, expiry, deadline, _ = f.factory.call_args.args
        self.assertEqual(len(pins), 7); self.assertEqual(release_pin, pin(seal.read_bytes()))
        self.assertEqual(expiry, NOW + 60); self.assertEqual(deadline, NOW - 100 + 7020)

    def test_paid_over150_unknown_negative_bool_and_nonfinite_rejected(self):
        cases = [("new_run_worst_case_usd", 30.01), ("verified_cumulative_spent_usd", None),
                 ("verified_cumulative_spent_usd", -1), ("other_reserved_max_usd", True),
                 ("unknown_or_pending_unbounded_charges", True)]
        for key, value in cases:
            with self.subTest(key=key, value=value):
                account = paid_account(); account[key] = value
                self.assert_denied(self.fixture(account), "PAID_NATIVE_CUMULATIVE_USD150")
        for value in (float("nan"), float("inf"), -float("inf"), True, None, -1):
            with self.subTest(finite=value): self.assertFalse(admission._finite(value))

    def test_kaggle_verified_quota_accepts_unknown_historic_spent_without_calling_it_zero(self):
        account = kaggle_account(); f = self.fixture(account)
        self.assertIs(f.invoke_inert(), f.inert_result)
        self.assertIsNone(f.docs["ACCOUNT-AND-BUDGET.json"]["verified_cumulative_spent_usd"])
        self.assertTrue(f.docs["ACCOUNT-AND-BUDGET.json"]["unknown_or_pending_unbounded_charges"])
        self.assertFalse(f.docs["ACCOUNT-AND-BUDGET.json"]["new_GCP_resources_or_cost"])

    def test_kaggle_no_incremental_GCP_verified_quota_mandatory(self):
        cases = [("new_GCP_resources_or_cost", True), ("new_GCP_resources_or_cost", None),
                 ("verified_incremental_cash_max_usd", .01), ("verified_incremental_cash_max_usd", False),
                 ("quota_and_no_incremental_charge_basis_verified", False),
                 ("GPU_quota_remaining_seconds", 7199.99), ("GPU_quota_remaining_seconds", None),
                 ("GPU_quota_remaining_seconds", True)]
        for key, value in cases:
            with self.subTest(key=key, value=value):
                account = kaggle_account(); account[key] = value
                self.assert_denied(self.fixture(account), "KAGGLE_VERIFIED_QUOTA_NO_NEW_GCP_COST_ONLY")

    def test_current_observation_expiry_and_one_hour_window(self):
        cases = [("observed_epoch", NOW + 1), ("valid_until_epoch", NOW),
                 ("valid_until_epoch", NOW + 3541), ("observed_epoch", None),
                 ("setup_anchor_epoch", True)]
        for key, value in cases:
            with self.subTest(key=key, value=value):
                account = paid_account(); account[key] = value
                self.assert_denied(self.fixture(account), "NATIVE_(CURRENT_ACCOUNT_PRICING|FINITE_BUDGET_ACCOUNT)")

    def test_original_setup_clock_cannot_restart_or_cross_science_deadline(self):
        for anchor in (NOW + .001, NOW - 7020, NOW - 7200):
            with self.subTest(anchor=anchor):
                account = paid_account(); account["setup_anchor_epoch"] = anchor
                self.assert_denied(self.fixture(account), "NATIVE_ORIGINAL_WHOLEWALL_7200_RESERVE180")

    def test_provider_denial_and_current_capacity_are_not_assumed(self):
        cases = [("provider_denial", True), ("provider_denial", None),
                 ("actual_accelerator_choice_verified", False), ("concurrent_account_capacity_verified", None),
                 ("gpu_count", 2), ("gpu_count", True), ("tpu_count", 1), ("tpu_count", False)]
        for key, value in cases:
            with self.subTest(key=key, value=value):
                account = kaggle_account(); account[key] = value
                self.assert_denied(self.fixture(account), "NATIVE_ONE_GPU_ACCOUNT_NOT_ASSUMED")

    def test_cumulative150_ceiling_cannot_be_raised_or_lowered(self):
        for budget in (149, 151, True, None):
            with self.subTest(budget=budget):
                account = paid_account(); account["total_cumulative_budget_usd"] = budget
                self.assert_denied(self.fixture(account), "NATIVE_(FINITE_BUDGET_ACCOUNT|CUMULATIVE_USD150)")

    def test_root_scope_bootstrap_and_exact_doc_set_fail_before_reads(self):
        for mutation, code in ((lambda f: f.grant.update(scope="PROBE_ONLY"), "NATIVE_EXPLICIT_ROOT_SCOPE"),
                               (lambda f: f.marker.update(mode="UNREVIEWED_IMPORT"), "REVIEWED_NATIVE_BYTE_BOOTSTRAP"),
                               (lambda f: f.grant["files"].pop(admission.NAMES[0]), "NATIVE_ROOT_DOC_SET")):
            f = self.fixture(); mutation(f)
            with mock.patch.object(Path, "read_bytes", side_effect=AssertionError("NO DOCUMENT READ BEFORE OUTER AUTHORITY")):
                self.assert_denied(f, code)

    def test_doc_byte_tampering_duplicate_json_and_symlink_rejected(self):
        f = self.fixture(); path = f.root / admission.NAMES[0]; path.write_bytes(path.read_bytes() + b" ")
        self.assert_denied(f, "NATIVE_ROOT_DOC_PIN")
        f = self.fixture(); raw = b'{"run_id":"SYNTHETIC","run_id":"DUPLICATE"}\n'
        (f.root / admission.NAMES[0]).write_bytes(raw); f.grant["files"][admission.NAMES[0]] = pin(raw)
        self.assert_denied(f, "DUPLICATE_KEY")
        f = self.fixture(); path = f.root / admission.NAMES[0]; moved = f.root / "SYNTHETIC_MOVED_DOC"
        path.rename(moved); path.symlink_to(moved)
        self.assert_denied(f, "NATIVE_DOC_REGULAR_BOUND")

    def test_science_scope_DEV_design_seen_and_failed_supply_gate_immutable(self):

    def test_dev_lineage_review_and_root_release_required(self):
        cases = [(admission.NAMES[2], "status", "PASS_SUPPLY_GO", "NATIVE_INDEPENDENT_SCIENCE_SOURCE_REVIEW"),
                 (admission.NAMES[4], "target_bodies_exposed_to_author", True, "NATIVE_DEV_SCORER_TARGET_LINEAGE_REQUIRED"),

        mutations = [(lambda d: d.pop("materializer_manifest_sha256"), "FROZEN_EXACT_DEV_ADMISSION_SCHEMA"),
                     (lambda d: d.update(unreviewed_extra="c" * 64), "FROZEN_EXACT_DEV_ADMISSION_SCHEMA"),

                     (lambda d: d.update(opaque_dev_goldpack_semantic_sha256=None), "FROZEN_DEV_COMPLETE_SHA256_PINS")]

    def test_DEV_rootbound_all_pins_commit_and_nonleakage_gates_must_agree(self):
        for key in sorted(admission.DEV_PIN_KEYS | {"official_commit"}):

        for key, value in (("no_correctness_before_192_and_model_stop", False),

                f.reseal_synthetic_docs(); self.assert_denied(f, "PRE_RUN_GOLDPACK_SCORER_TRANSFORM_EXACT_BINDING")
        for key in ("scorer_manifest_sha256", "public_dev_identity_semantic_sha256", "original_DEV32_lineage_map_sha256", "official_commit"):

    def test_existing_attempt_file_or_symlink_cannot_redirect_seal(self):
        for kind in ("file", "symlink"):
            with self.subTest(kind=kind):

class ReleasePathAndJournalTests(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory(prefix="SYNTHETIC_RELEASE_JOURNAL_"); self.addCleanup(self.temp.cleanup)
        self.root = Path(self.temp.name); self.attempt = self.root / "attempt"; self.attempt.mkdir(mode=0o700)
        # This is a method-test dummy, NOT a NativeRelease capability.
        self.dummy = SimpleNamespace(attempt_root=self.attempt)

    def test_output_bound_to_private_attempt_no_parent_escape_or_reuse(self):
        good = self.attempt / "SYNTHETIC_NEW.json"
        self.assertEqual(contracts.NativeRelease.assert_output_path(self.dummy, good), good)
        for bad in (Path("relative"), self.attempt, self.root / "outside", self.attempt / ".." / "escape"):
            with self.subTest(path=str(bad)), self.assertRaisesRegex(ValueError, "NATIVE_OUTPUT_MUST_BELONG"):
                contracts.NativeRelease.assert_output_path(self.dummy, bad)
        good.write_bytes(b"SYNTHETIC")
        with self.assertRaisesRegex(ValueError, "NATIVE_OUTPUT_EXCLUSIVE"):
            contracts.NativeRelease.assert_output_path(self.dummy, good)
        self.assertEqual(contracts.NativeRelease.assert_output_path(self.dummy, good, allow_existing=True), good)

    def test_output_symlink_ancestor_and_wrong_attempt_permissions_rejected(self):
        other = self.root / "SYNTHETIC_OTHER"; other.mkdir()
        link = self.attempt / "link"; link.symlink_to(other, target_is_directory=True)
        with self.assertRaisesRegex(ValueError, "NATIVE_OUTPUT_ANCESTOR_SYMLINK"):
            contracts.NativeRelease.assert_output_path(self.dummy, link / "output")
        self.attempt.chmod(0o755)
        with self.assertRaisesRegex(ValueError, "PRIVATE_ATTEMPT_MODE0700"):
            contracts.NativeRelease.assert_output_path(self.dummy, self.attempt / "output")

    def test_release_constructor_denies_PASS_dict(self):
        with self.assertRaisesRegex(ValueError, "ROOT_REVIEWED_NATIVE_RELEASE_FACTORY_REQUIRED"):
            contracts.NativeRelease((), "0" * 64, NOW + 1, NOW + 1, {"PASS": True})

    def _receipt_dummy(self):
        path = self.root / "SYNTHETIC_PUBLIC_RECEIPT.json"; raw = raw_json({"fixture": FIXTURE_RUN}); path.write_bytes(raw)
        synthetic_key = object()
        dummy = SimpleNamespace(_key=synthetic_key, document_paths_and_pins=((str(path), len(raw), pin(raw)),),
                                release_pin="0" * 64, valid_until_epoch=NOW + 10, science_deadline_epoch=NOW + 20)
        return dummy, synthetic_key, path

    def test_release_rechecks_expiry_before_read_and_public_byte_change(self):
        dummy, key, path = self._receipt_dummy()
        with mock.patch.object(contracts, "_RELEASE_KEY", key), mock.patch.object(contracts.time, "time", return_value=NOW):
            self.assertEqual(contracts.NativeRelease.verify_before_load(dummy), "0" * 64)
            path.write_bytes(path.read_bytes().replace(b"SYNTHETIC", b"XYNTHETIC", 1))
            with self.assertRaisesRegex(ValueError, "NATIVE_PUBLIC_RECEIPT_CHANGED"):
                contracts.NativeRelease.verify_before_load(dummy)
        for now in (NOW + 10, NOW + 20, NOW + 100):
            with mock.patch.object(contracts, "_RELEASE_KEY", key), mock.patch.object(contracts.time, "time", return_value=now), \
                 mock.patch.object(Path, "read_bytes", side_effect=AssertionError("expired release read")):
                with self.assertRaisesRegex(ValueError, "NATIVE_RELEASE_EXPIRED"):
                    contracts.NativeRelease.verify_before_load(dummy)

    def test_release_rejects_public_receipt_symlink_and_size_change(self):
        dummy, key, path = self._receipt_dummy()
        with mock.patch.object(contracts, "_RELEASE_KEY", key), mock.patch.object(contracts.time, "time", return_value=NOW):
            path.write_bytes(path.read_bytes() + b" ")
            with self.assertRaisesRegex(ValueError, "NATIVE_PUBLIC_RECEIPT_SIZE"):
                contracts.NativeRelease.verify_before_load(dummy)
            target = path.with_name("SYNTHETIC_TARGET.json"); path.rename(target); path.symlink_to(target)
            with self.assertRaisesRegex(ValueError, "NATIVE_PUBLIC_RECEIPT_PATH"):
                contracts.NativeRelease.verify_before_load(dummy)

    def test_private_journal_exclusive0600_fsync_and_exact_bytes(self):
        path = self.attempt / "SYNTHETIC_JOURNAL.jsonl"; journal = native_entry.PrivateJournal(path)
        self.addCleanup(journal.close)
        with mock.patch.object(native_entry.time, "time", return_value=NOW), mock.patch.object(native_entry.os, "fsync", wraps=os.fsync) as sync:
            journal.emit("SYNTHETIC_EVENT", {"synthetic": True, "n": 1}); sync.assert_called_once()
        self.assertEqual(stat.S_IMODE(path.stat().st_mode), 0o600)
        raw = path.read_bytes(); self.assertEqual(journal.bytes, len(raw)); self.assertEqual(journal.events, 1)
        self.assertEqual(json.loads(raw), {"event": "SYNTHETIC_EVENT", "fields": {"synthetic": True, "n": 1}, "epoch": NOW})
        with self.assertRaises(FileExistsError): native_entry.PrivateJournal(path)
        self.assertEqual(path.read_bytes(), raw); journal.close(); journal.close()

    def test_private_journal_leaf_symlink_never_followed(self):
        target = self.attempt / "SYNTHETIC_TARGET"; target.write_bytes(b"UNCHANGED")
        link = self.attempt / "SYNTHETIC_LINK"; link.symlink_to(target)
        with self.assertRaisesRegex(ValueError, "PRIVATE_JOURNAL_ABSOLUTE_NO_SYMLINKS"): native_entry.PrivateJournal(link)
        self.assertEqual(target.read_bytes(), b"UNCHANGED")

    def test_private_journal_bound_and_nonfinite_reject_before_write(self):
        path = self.attempt / "SYNTHETIC_BOUNDED.jsonl"; journal = native_entry.PrivateJournal(path); self.addCleanup(journal.close)
        with mock.patch.object(native_entry, "LOG_BYTES_RESERVATION", 1), \
             mock.patch.object(native_entry.os, "write", side_effect=AssertionError("over-budget write")):
            with self.assertRaisesRegex(ValueError, "PRIVATE_JOURNAL_DISK_RESERVATION"):

                      "engineering_public_probe": {"prompt": list(response.prompt),

                      "query_gold_used": False, "private_pairpack_used": False}}

                    self.assertEqual(len(reads), 2)
                    return b"SYNTHETIC_SENTINEL_NOT_A_PAIRPACK"

            def workflow(*args, **kwargs):
                kwargs["emit"]("SYNTHETIC_WORKFLOW_COMPLETE", {"fixture": True})
                if workflow_failure: raise RuntimeError("SYNTHETIC_FAIL_NO_RETRY")
                predictions = tuple({"arm": arm, "synthetic_ordinal": i}

                return {"synthetic_fixture_only": True}, predictions

    def test_all192_three_seals_journal_resets_and_private_permissions(self):
        status, reads, files, modes, parser, loader, loop = self.invoke_fixture()
        self.assertEqual(status["status"], "COMPLETE_PREDICTIONS_SEALED_WAIT_PROCESS_EXIT")
        self.assertEqual((len(reads), parser, loader, loop), (2,1,1,1))
        self.assertTrue(all(v == 0o600 for v in modes.values()))
        seal = json.loads(files["PREDICTIONS-SEALED.json"])
        self.assertTrue(seal["model_process_exit_still_required"])
        self.assertEqual(set(seal["distinct_arm_seals"]), {"fresh_base","CE","DPO"})
        self.assertEqual(len({v["sha256"] for v in seal["distinct_arm_seals"].values()}), 3)
        for key, name in (("private_journal", "PRIVATE-OPERATIONS-AND-PREDICTIONS.jsonl"),
                          ("private_resets", "PRIVATE-RESET-RECEIPTS.json"),
                          ("private_predictions", "PRIVATE-PREDICTIONS-192.json")):
            self.assertEqual(seal[key], {"bytes": len(files[name]), "sha256": hashlib.sha256(files[name]).hexdigest()})
        self.assertNotIn("MODEL-PROCESS-STOPPED.json", files)

    def test_bad_public_binding_stops_before_private_model_or_workflow(self):
        status, reads, files, modes, parser, loader, loop = self.invoke_fixture(
            mutation=lambda p,c: c.update(dev_public_panel_semantic_sha256="0"*64))
        self.assertEqual(status["private_pairpack_reads"], 0)
        self.assertEqual((len(reads), parser, loader, loop), (1,0,0,0))
        self.assertNotIn("PRIVATE-PREDICTIONS-192.json", files)

    def test_failed_engineering_blocks_workflow_after_one_private_fixture_read(self):
        status, reads, files, modes, parser, loader, loop = self.invoke_fixture(gate=False)
        self.assertEqual(status["status"], "FAILED_OR_UNKNOWN_ATTEMPT_CONSUMED")
        self.assertEqual((status["private_pairpack_reads"], parser, loader, loop), (1,1,1,0))
        self.assertNotIn("PREDICTIONS-SEALED.json", files)

    def test_wrong_device_stops_before_private_input_or_model_load(self):
        status, reads, files, modes, parser, loader, loop = self.invoke_fixture(device_failure=True)
        self.assertEqual(status["private_pairpack_reads"], 0)
        self.assertEqual((len(reads), parser, loader, loop), (1,0,0,0))
        self.assertNotIn("PRE-PRIVATE-PAIRPACK-SEAL.json", files)

    def test_workflow_failure_never_retries_or_seals_partial_predictions(self):
        status, reads, files, modes, parser, loader, loop = self.invoke_fixture(workflow_failure=True)
        self.assertEqual((len(reads), parser, loader, loop), (2,1,1,1))
        self.assertFalse(status["retry_allowed"])
        self.assertNotIn("PREDICTIONS-SEALED.json", files)
        self.assertNotIn("SYNTHETIC_FAIL_NO_RETRY", json.dumps(status))


class PreflightSourceReuseTests(unittest.TestCase):
    def exercise(self, callback):
        with tempfile.TemporaryDirectory(prefix="SYNTHETIC_TWO_STAGE_SOURCE_ONLY_") as tmp:
            root=Path(tmp); source=root/"fixture.py"; raw=b"SYNTHETIC_COMPILED_ONCE = object()\n"
            source.write_bytes(raw); pins={"cuda_backend":("fixture.py",hashlib.sha256(raw).hexdigest())}
            before=set(sys.modules)
            try:
                with mock.patch.object(backend_load,"SOURCE_PINS",pins),mock.patch.object(backend_load,"ASSET_PINS",{}):
                    sources=backend_load.verify_source_closure(root)
                    first=sources.module("cuda_backend")
                    release=mock.Mock(release_pin="7"*64)
                    cap=backend_load.NativeDevicePreflight(backend_load._BINDING_SENTINEL,sources,release,
                                                          {"release_sha256":release.release_pin})
                    callback(root,source,sources,first,release,cap)
            finally:
                for name in set(sys.modules)-before:
                    if name.startswith("_new_ps_verified_"):sys.modules.pop(name,None)

    def test_second_stage_rehash_reuses_exact_compiled_module_without_namespace_collision(self):
        def check(root,source,sources,first,release,cap):
            reused=backend_load._reuse_preflight_sources(cap,root,release)
            self.assertIs(reused,sources);self.assertIs(reused.module("cuda_backend"),first)
            self.assertIs(reused.module("cuda_backend").SYNTHETIC_COMPILED_ONCE,first.SYNTHETIC_COMPILED_ONCE)
        self.exercise(check)

    def test_source_changed_between_device_and_model_stages_denied(self):
        def check(root,source,sources,first,release,cap):
            source.write_bytes(b"SYNTHETIC_CHANGED=True\n")
            with self.assertRaisesRegex(ValueError,"source hash mismatch"):
                backend_load._reuse_preflight_sources(cap,root,release)
        self.exercise(check)

    def test_other_release_cannot_reuse_capability(self):
        def check(root,source,sources,first,release,cap):
            with self.assertRaisesRegex(ValueError,"same actual device"):
                backend_load._reuse_preflight_sources(cap,root,mock.Mock(release_pin=release.release_pin))
        self.exercise(check)

    def test_real_preflight_default_or_fake_release_denied_before_sources(self):
        with mock.patch.object(backend_load,"verify_source_closure",side_effect=AssertionError("no sources")):
            with self.assertRaisesRegex(ValueError,"default disabled"):
                backend_load.preflight_native_device(runtime_root="SYNTHETIC_NO_READ",release=object())
            with self.assertRaisesRegex(ValueError,"verified native release"):
                backend_load.preflight_native_device(runtime_root="SYNTHETIC_NO_READ",release=object(),enable_cuda_execution=True)

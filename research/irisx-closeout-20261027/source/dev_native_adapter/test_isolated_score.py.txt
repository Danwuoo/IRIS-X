class ScoreTests(unittest.TestCase):
    def setUp(self):
        self.temp=tempfile.TemporaryDirectory(prefix='synthetic-score-');self.addCleanup(self.temp.cleanup)
        self.root=Path(self.temp.name);self.attempt=self.root/'attempt';self.b,self.docs,self.gold=fixture(self.root)
        self.opens=0;self.evidence_reads=0
    def run_fixture(self,loader=None):
        def evidence(k,_pin):self.evidence_reads+=1;return self.docs[k]
        def gold(_p):
            self.assertTrue((self.attempt/'PRE-GOLD-SEAL.json').is_file());self.assertTrue((self.attempt/'GOLD-READ-INTENT.json').is_file())
            self.opens+=1;return self.gold if loader is None else loader()
        return sc.score_synthetic_once(self.attempt,lambda:self.b,evidence,gold)
    def test_complete_exactone_goldopen_afterseal_aggregate_only(self):
        r=self.run_fixture();self.assertEqual(r['status'],'COMPLETE_AGGREGATE_ONLY_NEW_RECONSTRUCTION');self.assertEqual(self.opens,1)
        self.assertEqual(r['goldpack_open_attempts'],1);self.assertFalse(r['quality_GO'])
        self.assertEqual(self.attempt.stat().st_mode&0o777,0o700)
        self.assertTrue(all(p.stat().st_mode&0o777==0o600 for p in self.attempt.iterdir()))
        aggregate=de.strict_json((self.attempt/'AGGREGATE-DEV32.json').read_bytes())
        self.assertNotIn('grid',aggregate);self.assertNotIn('rows',aggregate)
    def test_all_public_preflight_failures_zero_gold_evidenceopens(self):
        for key in ('contract','manifest','review','release','identity','context'):
            with self.subTest(key=key):
                self.attempt=self.root/key;old=self.b[key];self.b[key]=b'not JSON'
                r=self.run_fixture();self.assertEqual(r['status'],'FAILED_OR_UNKNOWN_ATTEMPT_CONSUMED_NO_RETRY')
                self.assertEqual(self.opens,0);self.assertEqual(self.evidence_reads,0);self.b[key]=old
    def test_source_mutation_zero_opens(self):
        self.b['sources']['dev_native_adapter/prediction_join.py']=b'changed'
        self.run_fixture();self.assertEqual(self.opens,0);self.assertEqual(self.evidence_reads,0)
    def test_mismatched_pred_bytes_zero_gold(self):
        self.docs['predictions']+=b' '
        self.run_fixture();self.assertEqual(self.opens,0)
    def test_missing_model_stop_zero_gold(self):
        self.docs['model_stop']=b'{}';self.run_fixture();self.assertEqual(self.opens,0)
    def test_incomplete_journal_zero_gold(self):

    def test_wrong_gold_hash_oneattempt_no_result(self):

class BootstrapTests(unittest.TestCase):
    def test_fresh_isolated_exact_byte_compile_no_native_entry(self):
        work=Path(__file__).resolve().parent.parent
        code="""from pathlib import Path
import types,sys
w=Path(sys.argv[1]);p=w/'dev_native_adapter/score_bootstrap.py'
m=types.ModuleType('bootstrap');m.__file__=str(p)
exec(compile(p.read_bytes(),str(p),'exec',dont_inherit=True),m.__dict__)
names=['dev_evaluator/dev_evaluator.py','dev_native_adapter/prediction_join.py','dev_native_adapter/isolated_score.py']
mods=m.compile_modules({n:(w/n).read_bytes() for n in names},'a'*64)
assert all(x._REVIEWED_TRANSFORM_MANIFEST_SHA256=='a'*64 for x in mods)
assert all(x.__file__.startswith(str(w)) for x in mods)
print('PASS_PUBLIC_SOURCE_COMPILE_NO_NATIVE_ENTRY')
"""
        r=subprocess.run([sys.executable,'-I','-S','-B','-c',code,str(work)],capture_output=True,text=True,timeout=10)
        self.assertEqual(r.returncode,0);self.assertEqual(r.stdout.strip(),'PASS_PUBLIC_SOURCE_COMPILE_NO_NATIVE_ENTRY')
    def test_root_loader_inert_and_private_exception_fixed(self):
        import score_root_loader as loader
        result=loader.run_with_root_grant({'private':'SYNTHETIC_SECRET'})
        self.assertEqual(result['status'],'FAILED_OR_UNKNOWN_NO_RETRY_NO_PRIVATE_ERROR_OUTPUT')
        self.assertNotIn('SYNTHETIC_SECRET',json.dumps(result))

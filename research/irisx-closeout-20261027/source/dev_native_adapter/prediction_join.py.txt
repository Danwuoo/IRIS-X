def sha(raw):return hashlib.sha256(raw).hexdigest()
def pin(raw):return {'bytes':len(raw),'sha256':sha(raw)}
def checkpin(expected,raw):
    de.closed(expected,{'bytes','sha256'},'ADAPTER_PIN_SCHEMA')
    de.need(type(expected['bytes']) is int and pin(raw)==expected,'ADAPTER_EXTERNAL_BYTE_PIN')

def finite_sum(values,code):
    try:result=math.fsum(values)
    except OverflowError:raise de.AdmissionError(code) from None
    de.need(math.isfinite(result),code);return result

def phase_kind(p):
    if p in [a+'_DEV_generate' for a in ARMS]:return 'generation'
    if p=='reference' or p in ('CE_fixed_prefix_KL','DPO_fixed_prefix_KL') or re.fullmatch(r'(CE|DPO)_train_[1-4]',p):return 'science'
    if p in ('CE_checkpoint_roundtrip_EXTRA_COST','DPO_checkpoint_roundtrip_EXTRA_COST'):return 'checkpoint'
    if p=='engineering_PUBLIC_ONLY_EXTRA_COST' or p in [a+'_DEV_parent_policy_replay_EXTRA_COST' for a in ARMS]:return 'engineering'
    if p in ('fresh_base_reset','reference_reset','CE_reset','DPO_reset','material_and_model_load'):return 'setup'
    raise de.AdmissionError('ADAPTER_UNDECLARED_PHASE')


def join_journal(raw,rows,*,public_prompt_lengths,native=None,run_id=None,release_sha256=None):
    de.need(type(public_prompt_lengths) is list and len(public_prompt_lengths)==32 and all(type(n) is int and 0<n<=2560 for n in public_prompt_lengths),'EXACT_PUBLIC_PROMPT_LENGTHS')
    de.need(type(raw) is bytes and 0<len(raw)<=256*1024**2 and raw.endswith(b'\n'),'SEALED_JSONL_BOUND')
    events=[de.strict_json(line) for line in raw.splitlines()]
    de.need(events and len(events)<=1000000,'JOURNAL_EVENT_BOUND')
    stack=[];root=None;draw=None;replay=None;draw_index=0;op_id=root_id=0
    roots=Counter();positions=Counter();ops=Counter();draw_info=[];extra_seconds=[];top_level_seconds=[];phase_events={};seal_count=0
    last_epoch=None;science_sequence=[];replay_keys=set();science_positions=0;science_backward_tokens=0;pending_science=[]
    for ordinal,item in enumerate(events):
        de.closed(item,{'event','fields','epoch'},'JOURNAL_CLOSED_EVENT')
        e,f,t=item['event'],item['fields'],item['epoch']
        de.need(e in EVENTS and type(f) is dict and type(t) in (int,float) and math.isfinite(t),'JOURNAL_KNOWN_FINITE_EVENT')
        # Wall time may be adjusted by the OS; source-monotonic operation elapsed
        # is authoritative for durations, not subtraction of journal timestamps.
        if last_epoch is None:last_epoch=t
        if 'phase' in f:
            kind=phase_kind(f['phase']);phase_events.setdefault(f['phase'],[]).append(ordinal)
        if e=='model_load_intent':
            de.closed(f,{'release_sha256','paid_or_quota_time_already_charged'},'MODEL_LOAD_SCHEMA')
            de.need(ordinal==0 and f['paid_or_quota_time_already_charged'] is True and (release_sha256 is None or f['release_sha256']==release_sha256),'MODEL_LOAD_FIRST')
        elif e=='draw_intent':
            de.closed(f,DRAW_KEYS,'DRAW_INTENT_SCHEMA');de.need(draw is None and not stack and root is None and replay is None,'EXCLUSIVE_DRAW_INTENT')
            de.need(draw_index<len(rows) and de.canonical(f)==de.canonical({k:rows[draw_index][k] for k in DRAW_KEYS}),'EXACT_SEALED_DRAW_ORDER')
            draw={'key':f,'operation':None,'elapsed':None,'physical':0}
        elif e=='draw_complete':
            de.need(draw is not None and not stack and root is None and de.canonical(f)==de.canonical(rows[draw_index]) and draw['elapsed'] is not None,'EXACT_COMPLETE_DRAW_JOIN')
            de.need(draw['physical']==len(rows[draw_index]['candidate']['sampled_logps'])+(6 if rows[draw_index]['status']=='VALID_EOS' else 5) and draw['physical']<=518,'EXACT_ORIGINAL_CHUNKED_GENERATION_PHYSICAL_CALLS')
            draw_info.append((draw['elapsed'],draw['physical']));draw_index+=1;draw=None
        elif e=='policy_replay_intent':
            de.need(replay is None and draw is None and not stack,'EXCLUSIVE_POLICY_REPLAY')
            de.closed(f,{'arm','ordinal'},'REPLAY_INTENT_SCHEMA');de.need(f['arm'] in ARMS and type(f['ordinal']) is int and 0<=f['ordinal']<32,'REPLAY_IDENTITY')
            de.need(draw_index<192 and draw_index%2==0 and (f['arm'],f['ordinal'])==(rows[draw_index]['arm'],rows[draw_index]['ordinal']) and (f['arm'],f['ordinal']) not in replay_keys,'EXACT_ONCE_REPLAY_BEFORE_PARENT_G2')
            replay_keys.add((f['arm'],f['ordinal']));replay=dict(f,physical=0)
        elif e=='policy_replay_complete':
            de.closed(f,{'arm','ordinal','receipt'},'REPLAY_COMPLETION_SCHEMA')
            de.need(type(f['ordinal']) is int and type(f['receipt']) is dict and replay is not None and not stack and root is None and all(f.get(k)==replay[k] for k in ('arm','ordinal')) and replay['physical']==12,'REPLAY_COMPLETES_INTENT');replay=None
        elif e=='operation_intent':
            event=f.get('event');de.need(event in OP_FIELDS,'CLOSED_ENGINE_OPERATION')
            de.closed(f,{'id','phase','event','native_cuda'}|OP_FIELDS[event],'OPERATION_INTENT_SCHEMA')
            de.need(type(f['id']) is int and f['id']==op_id+1 and type(f['native_cuda']) is bool,'MONOTONIC_OPERATION_INTENT')
            op_id+=1
            for key in ('input_positions','response_tokens','tracked_graphs','update_index','prompt_positions','response_cap','extra_public_probe_forwards','synthetic_response_tokens','extra_root_calls'):
                if key in f:de.need(type(f[key]) is int and f[key]>0,'OPERATION_EXACT_INTEGER_FIELDS')
            if 'grad' in f:de.need(type(f['grad']) is bool,'OPERATION_GRAD_BOOL')
            if 'public_synthetic_only' in f:de.need(f['public_synthetic_only'] is True,'ENGINEERING_PUBLIC_ONLY')
            de.need(native is None or f['native_cuda'] is native,'EXACT_NATIVE_VERSUS_SYNTHETIC_JOURNAL_PROFILE')
            if event=='forward':
                de.need(f['input_positions']<=3072 and f['response_tokens']<=512 and f['input_positions']>=f['response_tokens'],'FORWARD_CONTEXT_RESPONSE_CAP')
                if phase_kind(f['phase'])=='science':de.need(f['grad'] is bool(re.fullmatch(r'(CE|DPO)_train_[1-4]',f['phase'])),'SCIENCE_EXACT_GRAD_MODE')
            if event=='backward' and phase_kind(f['phase'])=='science':de.need(f['tracked_graphs']==(1 if f['phase'].startswith('CE_') else 2),'SCIENCE_EXACT_LIVE_GRAPH_COUNT')
            if event=='commit' and phase_kind(f['phase'])=='science':de.need(f['update_index']==int(f['phase'].rsplit('_',1)[1]),'SCIENCE_EXACT_UPDATE_INDEX')
            if event=='generation':
                de.need(draw is not None and draw['operation'] is None and not stack and f['phase']==draw['key']['arm']+'_DEV_generate' and f['response_cap']==512 and f['prompt_positions']==public_prompt_lengths[draw['key']['ordinal']],'ONE_GENERATION_PER_DRAW')
                draw['operation']=op_id
            if event=='generation_replay':de.need(replay is not None and not stack and f['extra_root_calls']==12 and f['synthetic_response_tokens']==6 and f['phase']==replay['arm']+'_DEV_parent_policy_replay_EXTRA_COST' and f['prompt_positions']==public_prompt_lengths[replay['ordinal']],'EXACT_REPLAY_OPERATION')
            stack.append((f,0))
        elif e=='operation_complete':
            de.need(bool(stack),'OPERATION_COMPLETION_REQUIRES_INTENT');start,depth=stack[-1]
            de.closed(f,set(start)|{'elapsed_seconds'},'OPERATION_COMPLETE_SCHEMA')
            elapsed=f['elapsed_seconds'];de.need(type(elapsed) in (int,float) and math.isfinite(elapsed) and elapsed>=0,'MONOTONIC_OPERATION_ELAPSED')
            de.need(de.canonical({k:f[k] for k in start})==de.canonical(start) and root is None,'EXACT_LIFO_OPERATION_COMPLETION')
            if f['event']=='forward':de.need(depth==1,'EXACT_ONE_ROOT_PER_TEACHER_FORCED_FORWARD')
            stack.pop();ops[(f['phase'],f['event'])]+=1
            if phase_kind(f['phase'])=='science':
                science_sequence.append((f['phase'],f['event']))
                if f['event']=='forward':
                    science_positions+=f['input_positions']
                    if f['grad']:pending_science.append((f['phase'],f['response_tokens']))
                elif f['event']=='backward':
                    de.need(len(pending_science)==f['tracked_graphs'] and all(p==f['phase'] for p,_ in pending_science),'SCIENCE_PENDING_GRAPH_PHASE_COUNT')
                    science_backward_tokens+=sum(n for _,n in pending_science);pending_science=[]
                elif f['event']=='commit':de.need(not pending_science,'NO_UNBACKWARDED_SCIENCE_AT_COMMIT')
            if f['event']=='generation':
                de.need(draw is not None and draw['operation']==f['id'],'GENERATION_MATCHED_DRAW');draw['elapsed']=float(elapsed)
            if not stack:
                top_level_seconds.append(float(elapsed))
                if phase_kind(f['phase']) in ('engineering','checkpoint'):extra_seconds.append(float(elapsed))
        elif e=='physical_forward_intent':
            de.closed(f,{'id','phase','input_positions','model_training','use_cache'},'PHYSICAL_INTENT_SCHEMA')
            de.need(root is None and type(f['id']) is int and f['id']==root_id+1 and type(f['input_positions']) is int and 0<f['input_positions']<=3072,'PHYSICAL_ID_CONTEXT')
            de.need(type(f['model_training']) is bool and type(f['use_cache']) is bool,'PHYSICAL_FLAGS')
            de.need(bool(stack) and stack[-1][0]['phase']==f['phase'],'ROOT_IN_EXACT_OPERATION_PHASE')
            operation,root_count=stack[-1];stack[-1]=(operation,root_count+1)
            if operation['event']=='forward':de.need(f['input_positions']==operation['input_positions'] and f['model_training'] is operation['grad'] and f['use_cache'] is False,'FORWARD_OWN_ROOT_POSITION_FLAGS_PARITY')
            root_id+=1;root=f
            if replay is not None:replay['physical']+=1
            if phase_kind(f['phase'])=='generation':
                de.need(draw is not None and len(stack)==1 and stack[-1][0]['event']=='generation' and stack[-1][0]['phase']==f['phase'],'GENERATION_ROOT_IN_EXACT_OPERATION')
                draw['physical']+=1
        elif e=='physical_forward_complete':
            de.closed(f,{'id','phase','input_positions'},'PHYSICAL_COMPLETE_SCHEMA')
            de.need(root is not None and de.canonical(f)==de.canonical({k:root[k] for k in f}),'EXACT_PHYSICAL_COMPLETION');roots[f['phase']]+=1;positions[f['phase']]+=f['input_positions'];root=None
        elif e=='predictions_seal':
            de.need(ordinal==len(events)-1 and draw_index==192 and not stack and root is None and draw is None and replay is None,'ALL192_BEFORE_FINAL_JOURNAL_SEAL')
            de.closed(f,{'run_id','prediction_count','semantic_sha256'},'JOURNAL_SEAL_SCHEMA')
            de.need((run_id is None or f['run_id']==run_id) and type(f['prediction_count']) is int and f.get('prediction_count')==192 and f.get('semantic_sha256')==de.digest(rows),'JOURNAL_SEALED_ROWS_HASH');seal_count+=1
    de.need(draw_index==192 and seal_count==1 and not stack and root is None and draw is None and replay is None,'COMPLETE_JOURNAL_NOT_UNKNOWN')
    expected={('reference','forward'):74,('CE_fixed_prefix_KL','forward'):74,('DPO_fixed_prefix_KL','forward'):74}
    for a in ('CE','DPO'):
        for update in range(1,5):
            p=f'{a}_train_{update}';expected[(p,'forward')]=37 if a=='CE' else 74;expected[(p,'backward')]=37;expected[(p,'commit')]=1
    observed={k:v for k,v in ops.items() if phase_kind(k[0])=='science'}
    de.need(observed==expected,'JOURNAL_FIXED_666_TF_296_BACKWARD_8_COMMITS')
    ordered=[('reference','forward')]*74
    for a in ('CE','DPO'):
        for u in range(1,5):
            phase=f'{a}_train_{u}'
            ordered+=([(phase,'forward')]*(1 if a=='CE' else 2)+[(phase,'backward')])*37+[(phase,'commit')]
        ordered +=[(a+'_fixed_prefix_KL','forward')]*74
    de.need(science_sequence==ordered and all(roots[p]==v for (p,event),v in expected.items() if event=='forward'),'EXACT_ORDERED_SCIENCE_OPERATIONS_AND_ROOTS')
    de.need(len(replay_keys)==96 and all(ops[(a+'_DEV_parent_policy_replay_EXTRA_COST','generation_replay')]==32 for a in ARMS),'EXACT96_ORIGINAL_REPLAY_PROBES')
    de.need(all(ops[(a+'_DEV_generate','generation')]==64 for a in ARMS),'ALL64_GENERATION_OPERATIONS')
    return {'draws':draw_info,'root_by_phase':dict(roots),'positions_by_phase':dict(positions),'root_total':root_id,'operation_total':op_id,
      'science_forward_input_positions':science_positions,'science_backward_response_tokens':science_backward_tokens,'phase_event_ordinals':phase_events,'extra_seconds':finite_sum(extra_seconds,'EXTRA_SECONDS_FINITE'),'disjoint_top_level_operation_seconds':finite_sum(top_level_seconds,'ALL_DISJOINT_OPERATION_SECONDS_FINITE'),'first_epoch':events[0]['epoch'],'maximum_epoch':max(e['epoch'] for e in events)}


def normalize(raw_docs,expected_pins,identity_document,context,*,expected_context_sha256):
    """Target-free transform; external caller owns one-shot/private I/O admission.

    Receipt fingerprints are not authority. Exact source/review/root/private-read
    admission must be established outside this no-I/O function before calling it.
    """
    de.closed(raw_docs,KEYS,'ADAPTER_INPUT_BYTES');de.closed(expected_pins,KEYS,'ADAPTER_INPUT_PINS')
    for k in KEYS:checkpin(expected_pins[k],raw_docs[k])
    de.need(de.digest(context)==expected_context_sha256,'EXTERNAL_ADAPTER_CONTEXT_PIN')
    de.closed(context,{'schema','run_id','native_contract_sha256','native_source_sha256','native_release_sha256',
      'identity_sha256','bindings','wrapper_source_manifest_sha256','wrapper_contract_sha256','setup_anchor_epoch','historical_acquisition_seconds','opaque_materialization_receipt_sha256'},'ADAPTER_CONTEXT_SCHEMA')
    de.need(context['schema']=='new-ps-native-prediction-normalization-context/v1','ADAPTER_CONTEXT_VERSION')
    de.need(all(de.sha_ok(context[k]) for k in ('native_contract_sha256','native_source_sha256','native_release_sha256','identity_sha256','wrapper_source_manifest_sha256','wrapper_contract_sha256','opaque_materialization_receipt_sha256')),'EXACT_CONTEXT_HASH_TYPES')
    de.need(context['bindings']['stage_source_sha256']==context['native_source_sha256'],'STAGE_SOURCE_BINDING_EQUALS_NATIVE_CLOSURE')
    ident=de.validate_identity(identity_document,expected_sha256=context['identity_sha256'])
    docs={k:de.strict_json(v) for k,v in raw_docs.items() if k!='journal'}
    pred=docs['predictions'];de.closed(pred,{'schema','contract_sha256','rows'},'NATIVE_192_SCHEMA')
    de.need(pred['schema']=='new-ps-native-192-predictions/v1' and pred['contract_sha256']==context['native_contract_sha256'],'NATIVE_PREDICTION_CONTRACT')
    rows=pred['rows'];de.need(type(rows) is list and len(rows)==192,'EXACT_NATIVE192_NO_FILTER')
    policies={}
    for index,row in enumerate(rows):
        de.closed(row,DRAW_KEYS|{'candidate','status'},'NATIVE_DRAW_SCHEMA')
        arm=ARMS[index//64];ordinal=(index%64)//2;slot=index%2;p=identity_document['rows'][ordinal]
        de.need(row['arm']==arm and type(row['ordinal']) is int and row['ordinal']==ordinal and type(row['slot']) is int and row['slot']==slot
          and type(row['query_demo_ordinal']) is int and row['query_demo_ordinal']==p['query_demo_ordinal'] and row['parent_id']==p['parent_id']
          and type(row['seed']) is int and row['seed']==1729+ordinal*4096+slot and row['view_id']=='identity','EXACT_ORIGINAL_ROW_ORDER_SEED_FOLD')
        de.need(de.sha_ok(row['policy_adapter_sha256']),'NATIVE_ADAPTER_SHA')
        policies.setdefault(arm,row['policy_adapter_sha256']);de.need(policies[arm]==row['policy_adapter_sha256'],'FROZEN_POLICY_PER_ARM')
        de.closed(row['candidate'],{'grid','finished_eos','truncated','sampled_logps'},'NATIVE_CANDIDATE_SCHEMA')
        de.need(row['status'] in ('VALID_EOS','KNOWN_INVALID_CAP'),'NATIVE_UNKNOWN_NOT_INVALID')
    joint=docs['joint_seal'];de.closed(joint,{'schema','run_id','draws','private_predictions','aggregate','private_journal','private_resets','distinct_arm_seals',
      'contract_sha256','source_manifest_sha256','all_physical_intents_terminal','DEV_target_reads','model_process_exit_still_required'},'JOINT_SEAL_SCHEMA')
    de.need(joint['schema']=='new-ps-native-predictions-seal/v1' and joint['run_id']==context['run_id'] and type(joint['draws']) is int and joint['draws']==192
      and joint['contract_sha256']==context['native_contract_sha256'] and joint['source_manifest_sha256']==context['native_source_sha256']
      and joint['all_physical_intents_terminal'] is True and type(joint['DEV_target_reads']) is int and joint['DEV_target_reads']==0
      and joint['model_process_exit_still_required'] is True,'JOINT_EXACT_SCOPE')
    for key,doc in (('private_predictions','predictions'),('aggregate','aggregate'),('private_journal','journal'),('private_resets','resets')):
        de.need(joint[key]==expected_pins[doc],'JOINT_EXACT_BYTE_EVIDENCE')
    de.closed(joint['distinct_arm_seals'],ARMS,'THREE_ACTUAL_ARM_SEALS')
    for a in ARMS:
        seal=docs['seal_'+a]
        de.closed(seal,{'schema','run_id','arm','draws','ordered_rows_semantic_sha256','joint_private_prediction_file','private_journal','private_resets',
          'release_sha256','contract_sha256','all192_complete_before_arm_seal','DEV_target_reads'},'ARM_SEAL_SCHEMA')
        de.need(type(seal['draws']) is int and type(seal['DEV_target_reads']) is int and seal['all192_complete_before_arm_seal'] is True,'ARM_SEAL_EXACT_TYPES')
        de.need(seal=={'schema':'new-ps-native-arm-predictions-seal/v1','run_id':context['run_id'],'arm':a,'draws':64,
          'ordered_rows_semantic_sha256':de.digest([r for r in rows if r['arm']==a]),'joint_private_prediction_file':expected_pins['predictions'],
          'private_journal':expected_pins['journal'],'private_resets':expected_pins['resets'],'release_sha256':context['native_release_sha256'],
          'contract_sha256':context['native_contract_sha256'],'all192_complete_before_arm_seal':True,'DEV_target_reads':0}
          and joint['distinct_arm_seals'][a]==expected_pins['seal_'+a],'ACTUAL_ARM_SEAL_CROSS_BINDING')
    stop=docs['model_stop'];de.closed(stop,{'schema','run_id','child_pid','observed_returncode','wait_returned','deadline_termination','model_process_stopped',
      'observed_epoch','supervisor_communication_failure','supervisor_termination_signal','parent_death_guard','host_or_provider_failure_is_unknown',
      'contract_sha256','DEV_target_reads_by_supervisor','retry_allowed','scoring_admitted','predictions_seal','status','pre_setup_clock'},'MODEL_STOP_SCHEMA')
    de.need(stop['pre_setup_clock']==expected_pins['pre_setup_clock'],'MODEL_STOP_ORIGINAL_FIRST_CLOCK_PIN')
    de.need(type(stop['child_pid']) is int and stop['child_pid']>0,'MODEL_STOP_PID_INTEGER')
    de.need(stop['schema']=='new-ps-native-model-process-stopped/v1' and stop['run_id']==context['run_id'] and stop['contract_sha256']==context['native_contract_sha256']
      and stop['predictions_seal']==expected_pins['joint_seal'] and stop['wait_returned'] is True and stop['model_process_stopped'] is True
      and type(stop['observed_returncode']) is int and stop['observed_returncode']==0 and stop['deadline_termination'] is False and stop['scoring_admitted'] is False
      and stop['supervisor_communication_failure'] is False and stop['supervisor_termination_signal'] is None
      and stop['parent_death_guard']=='LINUX_PR_SET_PDEATHSIG_SIGKILL_EXPECTED_PARENT_RACECHECK' and stop['host_or_provider_failure_is_unknown'] is True
      and stop['retry_allowed'] is False and type(stop['DEV_target_reads_by_supervisor']) is int and stop['DEV_target_reads_by_supervisor']==0
      and stop['status']=='MODEL_STOPPED_SEALED_PREDICTIONS_AWAIT_SEPARATE_SCORER_ADMISSION','OBSERVED_SUCCESSFUL_PROCESS_EXIT_REQUIRED')
    outer=docs['wrapper_terminal'];export=docs['export_receipt']
    de.closed(outer,{'schema','status','run_id','setup_anchor_epoch','finished_epoch','wall_from_immutable_anchor_seconds','outer_entry_epoch',
      'outer_elapsed_monotonic_seconds','end_boundary','native_contract_sha256','native_source_manifest_sha256','wrapper_contract_sha256',
      'wrapper_source_manifest_sha256','native_release_sha256','actual_setup_epoch','actual_setup_monotonic','finished_monotonic',
      'setup_to_export_monotonic_seconds','prepublication_budget_offset_seconds','budget_elapsed_seconds','pre_setup_clock_receipt',
      'controller_exit_observed','controller_returncode','export_receipt','model_stop_receipt','retry_allowed','provider_resource_release','correctness_scoring_performed','quality_GO'},'WHOLE_WRAPPER_TERMINAL_SCHEMA')
    de.need(outer['schema']=='new-ps-whole-wrapper-terminal/v1' and outer['status']=='COMPLETE_FIXED37_ALL192_SMALL_EXPORTS' and outer['run_id']==context['run_id'] and
      outer['native_contract_sha256']==context['native_contract_sha256'] and outer['native_release_sha256']==context['native_release_sha256'] and outer['native_source_manifest_sha256']==context['native_source_sha256'] and
      de.sha_ok(context['wrapper_source_manifest_sha256']) and outer['wrapper_source_manifest_sha256']==context['wrapper_source_manifest_sha256'] and
      de.sha_ok(context['wrapper_contract_sha256']) and outer['wrapper_contract_sha256']==context['wrapper_contract_sha256'] and
      outer['model_stop_receipt']==expected_pins['model_stop'] and outer['export_receipt']==expected_pins['export_receipt'] and
      outer['controller_exit_observed'] is True and type(outer['controller_returncode']) is int and outer['controller_returncode']==0 and
      outer['pre_setup_clock_receipt']==expected_pins['pre_setup_clock'] and outer['retry_allowed'] is False and outer['provider_resource_release']=='UNKNOWN' and outer['correctness_scoring_performed'] is False and outer['quality_GO'] is False,
      'EXACT_OUTER_EXPORT_NATIVE_STOP_SOURCE_CLOSURE')
    de.need(outer['end_boundary']=='AFTER_DURABLE_EXPORT_RECEIPT_AND_EXPORT_CHILD_EXIT_BEFORE_THIS_TERMINAL_BOOKKEEPING','FULL_SETUP_THROUGH_EXPORT_BOUNDARY')

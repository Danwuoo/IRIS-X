def sha(b):return hashlib.sha256(b).hexdigest()
def pin(b):return {'bytes':len(b),'sha256':sha(b)}
def no_links(p):de.need(p.is_absolute() and not any(x.is_symlink() for x in (p,*p.parents)),'SCORE_NO_SYMLINK_ANCESTOR')
def syncdir(p):
    fd=os.open(p,os.O_RDONLY|os.O_DIRECTORY)
    try:os.fsync(fd)
    finally:os.close(fd)
def write_once(p,obj):
    raw=de.canonical(obj)+b'\n';fd=os.open(p,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
    with os.fdopen(fd,'wb') as f:f.write(raw);f.flush();os.fsync(f.fileno())
    syncdir(p.parent);return pin(raw)
def claim(p):

def read_public(p):

def read_private(p,cap):
    no_links(p);de.need(p.parent.is_dir() and p.parent.stat().st_mode&0o777==0o700,'PRIVATE_INPUT_PARENT0700')
    fd=os.open(p,os.O_RDONLY|os.O_NOFOLLOW)
    with os.fdopen(fd,'rb') as f:

def _run(attempt,bundle_loader,evidence_loader,gold_loader,mode,grant=None):
    opens=0;start=time.monotonic();old=None
    try:
        de.need(signal.getitimer(signal.ITIMER_REAL)==(0.0,0.0),'FRESH_SCORE_PROCESS_TIMER')
        old=signal.signal(signal.SIGALRM,lambda *_:(_ for _ in ()).throw(de.AdmissionError('SCORING_DEADLINE')))
        signal.setitimer(signal.ITIMER_REAL,60)
        b=bundle_loader();c,identity,ctx=preflight(b,mode,grant)
        raw={k:evidence_loader(k,x) for k,x in c['evidence'].items()}
        private_bundle,metadata=pj.normalize(raw,{k:x['pin'] for k,x in c['evidence'].items()},identity,ctx,expected_context_sha256=de.digest(ctx))
        ident=de.validate_identity(identity,expected_sha256=c['identity_semantic_sha256'])
        predictions=de.validate_predictions(private_bundle,ident,expected_sha256=de.digest(private_bundle),expected_bindings=ctx['bindings'],expected_closure_sha256=de.digest(private_bundle['closure']))
        de.need(predictions.complete and metadata['all_predictions_complete'] is True,'COMPLETE_NATIVE_EVIDENCE_BEFORE_GOLD')
        seal=write_once(attempt/'PRE-GOLD-SEAL.json',{'schema':'new-ps-native-dev-pre-gold-seal/v1','run_id':c['run_id'],
          'contract_sha256':sha(b['contract']),'source_manifest_sha256':sha(b['manifest']),'review_sha256':sha(b['review']),
          'release_sha256':sha(b['release']),'prediction_bundle_semantic_sha256':de.digest(private_bundle),
          'closure_semantic_sha256':de.digest(private_bundle['closure']),'model_stop_sha256':c['evidence']['model_stop']['pin']['sha256'],
          'goldpack_sha256':c['goldpack_sha256'],'goldpack_semantic_sha256':c['goldpack_semantic_sha256'],
          'exact192_complete_model_stopped':True,'goldpack_opens':0,'retry_allowed':False})
        write_once(attempt/'GOLD-READ-INTENT.json',{'schema':'new-ps-native-dev-gold-read-intent/v1','goldpack_sha256':c['goldpack_sha256'],'maximum_opens':1,'failure_consumes_attempt':True})
        opens=1;gold_raw=gold_loader(c['goldpack_path'])
        de.need(type(gold_raw) is bytes and 0<len(gold_raw)<=32*1024**2 and sha(gold_raw)==c['goldpack_sha256'],'EXACT_GOLD_BYTES_AFTER_ONE_OPEN')
        target=de.strict_json(gold_raw)
        case=de._admit_private_target_document(predictions,target,expected_target_sha256=c['goldpack_semantic_sha256'])
        aggregate=de._evaluate_admitted_case(case)
        aggregate_pin=write_once(attempt/'AGGREGATE-DEV32.json',aggregate)
        score_seconds=time.monotonic()-start
        result={'schema':'new-ps-isolated-dev-score-result/v1','status':'COMPLETE_AGGREGATE_ONLY_NEW_RECONSTRUCTION',
          'goldpack_open_attempts':opens,'aggregate_sha256':aggregate_pin['sha256'],'pre_gold_seal_sha256':seal['sha256'],
          'source_manifest_sha256':sha(b['manifest']),'isolated_cpu_scoring_seconds':score_seconds,
          'cost_scope':'MODEL_SETUP_THROUGH_SMALL_EXPORTS_PLUS_SEPARATE_ISOLATED_CPU_SCORING',
          'summed_known_pipeline_seconds_including_acquisition_and_scoring':aggregate['cost']['total_pipeline_seconds_including_sunk_acquisition']+score_seconds,
          'model_provider_GPU_calls':0,'author_per_parent_label_output':False,'quality_GO':False,'retry_allowed':False}
        write_once(attempt/'ATTEMPT-RESULT.json',result);return result
    except BaseException:
        result={'schema':'new-ps-isolated-dev-score-result/v1','status':'FAILED_OR_UNKNOWN_ATTEMPT_CONSUMED_NO_RETRY',
          'goldpack_open_attempts':opens,'model_provider_GPU_calls':0,'author_per_parent_label_output':False,'quality_GO':False,'retry_allowed':False}
        try:write_once(attempt/'ATTEMPT-RESULT.json',result)
        except BaseException:pass
        return result
    finally:
        if old is not None:signal.setitimer(signal.ITIMER_REAL,0);signal.signal(signal.SIGALRM,old)
